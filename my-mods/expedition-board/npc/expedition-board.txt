// expedition-board -- Contratos de expediciones.
//
// Las lineas de cabecera usan TAB entre campos, no espacios.
//
// Estado por personaje (sin prefijo de ambito = permanente en char_reg_num):
//   eb_qid    contrato activo (1..N), 0 si no hay
//   eb_state  0 libre, 1 cazando, 2 marca abatida, listo para entregar
//   eb_gid    GID de la marca invocada; invalido tras reiniciar el map-server
//
// La marca usa el evento ExpeditionCtrl::OnHuntDead, de modo que solo MUERE
// ESE spawn completa el contrato, no cualquier monstruo de la misma especie.

function	script	F_Expedition_Place	{
	.@map$ = getarg(0);
	.@x = getarg(1);
	.@y = getarg(2);
	if (checkcell(.@map$, .@x, .@y, cell_chkpass)) {
		set getvariableofnpc(.place_x, "ExpeditionCtrl"), .@x;
		set getvariableofnpc(.place_y, "ExpeditionCtrl"), .@y;
		return 1;
	}
	for (.@r = 1; .@r <= 24; .@r++) {
		for (.@dx = -.@r; .@dx <= .@r; .@dx++) {
			for (.@dy = -.@r; .@dy <= .@r; .@dy++) {
				if (abs(.@dx) != .@r && abs(.@dy) != .@r)
					continue;
				.@nx = .@x + .@dx;
				.@ny = .@y + .@dy;
				if (!checkcell(.@map$, .@nx, .@ny, cell_chkpass))
					continue;
				set getvariableofnpc(.place_x, "ExpeditionCtrl"), .@nx;
				set getvariableofnpc(.place_y, "ExpeditionCtrl"), .@ny;
				return 1;
			}
		}
	}
	debugmes "expedition-board: no walkable cell near " + .@map$ + " " + .@x + "," + .@y;
	return 0;
}

function	script	F_Expedition_ClearMark	{
	if (eb_gid > 0 && unitexists(eb_gid))
		unitkill eb_gid;
	eb_gid = 0;
	return;
}

function	script	F_Expedition_Spawn	{
	.@qid = getarg(0);
	if (.@qid < 1 || .@qid > getvariableofnpc(.count, "ExpeditionCtrl"))
		return 0;

	callfunc "F_Expedition_ClearMark";

	.@map$ = getvariableofnpc(.map$[.@qid], "ExpeditionCtrl");
	.@x = getvariableofnpc(.x[.@qid], "ExpeditionCtrl");
	.@y = getvariableofnpc(.y[.@qid], "ExpeditionCtrl");
	.@mob = getvariableofnpc(.mob[.@qid], "ExpeditionCtrl");
	if (!callfunc("F_Expedition_Place", .@map$, .@x, .@y))
		return 0;

	.@sx = getvariableofnpc(.place_x, "ExpeditionCtrl");
	.@sy = getvariableofnpc(.place_y, "ExpeditionCtrl");
	eb_gid = monster(.@map$, .@sx, .@sy, "Marca de expedicion", .@mob, 1, "ExpeditionCtrl::OnHuntDead");
	if (eb_gid < 1) {
		debugmes "expedition-board: monster() failed for qid " + .@qid + " on " + .@map$;
		return 0;
	}
	return 1;
}

		end;

	.@map$ = getvariableofnpc(.map$[eb_qid], "ExpeditionCtrl");
	if (killedrid != .@mob)
		end;

	.@count = getvariableofnpc(.count, "ExpeditionCtrl");

	mes "Avisos de caza para quien pueda salir de la ciudad y volver.";

		mes " ";

		mes "Marca en ^0000FF" + getvariableofnpc(.map$[eb_qid], "ExpeditionCtrl") + " (" + getvariableofnpc(.x[eb_qid], "ExpeditionCtrl") + ", " + getvariableofnpc(.y[eb_qid], "ExpeditionCtrl") + ")^000000.";

	} else if (eb_state == 2 && eb_qid >= 1 && eb_qid <= .@count) {

			mes "Vuelve al tablero de ^0000FF" + getvariableofnpc(.cityname$[eb_qid], "ExpeditionCtrl") + "^000000.";

	.@n = 0;
