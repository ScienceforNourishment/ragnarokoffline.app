// expedition-board -- Contratos de expediciones y cacerias.
//
// Las lineas de cabecera usan TAB entre campos, no espacios.
// Los contratos viven en expedition-contracts.txt (generado desde un CSV).
// Hay dos tipos (.type en los datos):
//   1 expedicion  una marca (monstruo con nombre) espera en un punto
//   2 caceria     hay que abatir N monstruos de una especie dentro de un mapa
//
// Donde se entrega cada tipo (siempre en la ciudad que publico el contrato):
//   expedicion  con el Intendente, un NPC aparte junto al tablero
//   caceria     en el propio tablero
// F_Expedition_AtBoard(type) lo decide; para mover una entrega a otro NPC u otra
// ciudad basta con tocar eso y colocar al NPC que llame a F_Expedition_Pay.
//
// Estado por personaje (sin prefijo de ambito = permanente en char_reg_num):
//   eb_qid       id del contrato activo, 0 si no hay
//   eb_state     0 libre, 1 en curso, 2 cumplido, listo para entregar
//   eb_prog      caceria: monstruos abatidos hasta ahora
//   eb_hunt_mob  caceria en curso: id del monstruo a contar; 0 si no hay. Cuenta
//                tambien cualquier otro id con su mismo nombre (S_Hunt). Es lo
//                primero que mira OnNPCKillEvent, que corre con CADA muerte
//   eb_hunt_map$ caceria en curso: el mapa donde cuenta
//   eb_gid       en desuso (la primera version guardaba aqui el GID); se pone a 0
//
// Estado temporal del servidor (se pierde al reiniciar, a proposito):
//   $@eb_mark[char_id]  GID de la marca viva de ese personaje (solo expediciones)
//   $@eb_markowner[gid] char_id del dueno de esa marca; sirve para acreditar al
//                       dueno si la abate alguien de su grupo
//   $@eb_hunters        cuantos personajes conectados tienen una caceria en
//                       curso; con 0 no se mira el grupo en cada muerte
//
// Credito compartido: si la marca la abate un miembro del grupo del dueno, o
// una caceria suma por la muerte de un companero de grupo, se acredita (ver
// OnMarkDead y OnNPCKillEvent). Hace falta que el otro este conectado y en el
// mismo mapa; fuera de un grupo solo cuenta lo propio.
//
// La marca se invoca con el evento ExpeditionCtrl::OnMarkDead, de modo que solo
// MUERE ESE spawn completa el contrato, no cualquier monstruo de la misma
// especie. Un GID guardado en el personaje podria apuntar, tras un reinicio, a
// otro monstruo; por eso vive en una variable temporal.

// F_Expedition_MarkAlive() -> 1 si el personaje tiene su marca en pie.
function	script	F_Expedition_MarkAlive	{
	.@gid = $@eb_mark[getcharid(0)];
	return (.@gid > 0 && unitexists(.@gid));
}

// F_Expedition_Tier(id) -> Tier de peligro (1-6) del mapa del contrato segun
// world-atlas; 0 si el atlas no esta activo o no conoce el mapa. world-atlas es
// una dependencia suave: $@wa_ready solo existe cuando su tabla esta llena.
function	script	F_Expedition_Tier	{
	if (!$@wa_ready)
		return 0;
	.@id = getarg(0);
	return callfunc("F_Atlas_Tier", getvariableofnpc(.map$[.@id], "ExpeditionData"));
}

// F_Expedition_Zeny(id) -> zeny que paga el contrato: el del CSV, +25% por cada
// Tier por encima del 1. Sin Tier (0) o en Tier 1 paga lo del CSV.
function	script	F_Expedition_Zeny	{
	.@id = getarg(0);
	.@zeny = getvariableofnpc(.zeny[.@id], "ExpeditionData");
	.@tier = callfunc("F_Expedition_Tier", .@id);
	if (.@tier > 1)
		.@zeny = .@zeny * (100 + 25 * (.@tier - 1)) / 100;
	return .@zeny;
}

// F_Expedition_Tag(id) -> linea de peligro para el dialogo, "" si no hay Tier.
function	script	F_Expedition_Tag	{
	.@tier = callfunc("F_Expedition_Tier", getarg(0));
	if (.@tier < 1)
		return "";
	return "Peligro: ^FF0000Tier " + .@tier + "^000000 de 6.";
}

// F_Expedition_Goal(id) -> el objetivo del contrato, en una linea de dialogo.
function	script	F_Expedition_Goal	{
	.@id = getarg(0);
	.@place$ = getvariableofnpc(.place$[.@id], "ExpeditionData");
	.@map$ = getvariableofnpc(.map$[.@id], "ExpeditionData");
	if (getvariableofnpc(.type[.@id], "ExpeditionData") == 2) {
		.@count = getvariableofnpc(.count[.@id], "ExpeditionData");
		.@mob = getvariableofnpc(.mob[.@id], "ExpeditionData");
		return "Abatir ^0000FF" + .@count + " " + getmonsterinfo(.@mob, MOB_NAME) + "^000000 en ^0000FF" + .@place$ + "^000000 (" + .@map$ + "). Solo cuentan en ese mapa.";
	}
	.@x = getvariableofnpc(.x[.@id], "ExpeditionData");
	.@y = getvariableofnpc(.y[.@id], "ExpeditionData");
	return "Marca en ^0000FF" + .@place$ + "^000000 (" + .@map$ + " " + .@x + ", " + .@y + ").";
}

// F_Expedition_AtBoard(type) -> 1 si ese tipo de contrato se entrega en el tablero;
// 0 si se entrega con el Intendente.
function	script	F_Expedition_AtBoard	{
	return (getarg(0) == 2);
}

// F_Expedition_Handoff(id) -> a quien se entrega, en una frase de dialogo.
function	script	F_Expedition_Handoff	{
	.@id = getarg(0);
	.@city$ = getvariableofnpc(.cityname$[.@id], "ExpeditionData");
	if (callfunc("F_Expedition_AtBoard", getvariableofnpc(.type[.@id], "ExpeditionData")))
		return "Entrega en el tablero de ^0000FF" + .@city$ + "^000000.";
	return "Entrega con el ^0000FFIntendente de " + .@city$ + "^000000, junto al tablero.";
}

// F_Expedition_Pay(title$) -> cierra el contrato activo y paga. Escribe su parte
// del dialogo (con el titulo dado) y devuelve 1 si pago, 0 si no cabia el peso.
// El estado se cierra antes de pagar y solo si cabe la recompensa.
function	script	F_Expedition_Pay	{
	.@qid = eb_qid;
	.@item = getvariableofnpc(.item[.@qid], "ExpeditionData");
	.@amt = getvariableofnpc(.amt[.@qid], "ExpeditionData");
	.@zeny = callfunc("F_Expedition_Zeny", .@qid);
	mes getarg(0);
	if (!checkweight(.@item, .@amt)) {
		mes "No puedes cargar la recompensa. Libera peso y vuelve.";
		return 0;
	}
	callfunc "F_Expedition_Reset";
	Zeny += .@zeny;
	getitem .@item, .@amt;
	mes "Contrato cerrado. " + .@zeny + " zeny y " + .@amt + " " + getitemname(.@item) + ".";
	return 1;
}

// F_Expedition_PartyAid(party id, char id) -> account id de ese miembro del grupo
// si esta conectado; 0 si no. Es lo que hace falta para attachrid.
function	script	F_Expedition_PartyAid	{
	getpartymember getarg(0), 1;
	.@n = $@partymembercount;
	if (.@n < 1)
		return 0;
	copyarray .@cid[0], $@partymembercid[0], .@n;
	getpartymember getarg(0), 2;
	copyarray .@aid[0], $@partymemberaid[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@cid[.@i] == getarg(1))
			return .@aid[.@i];
	}
	return 0;
}

// F_Expedition_HuntEnd() -> el personaje deja de contar una caceria (terminada o
// abandonada) y sale del recuento de $@eb_hunters.
function	script	F_Expedition_HuntEnd	{
	if (eb_hunt_mob) {
		if ($@eb_hunters > 0)
			$@eb_hunters--;
	}
	eb_hunt_mob = 0;
	eb_hunt_map$ = "";
	return;
}

// F_Expedition_ClearMark() -> retira la marca del personaje, si existe.
function	script	F_Expedition_ClearMark	{
	.@gid = $@eb_mark[getcharid(0)];
	$@eb_mark[getcharid(0)] = 0;
	if (.@gid > 0)
		$@eb_markowner[.@gid] = 0;
	if (.@gid > 0 && unitexists(.@gid))
		unitkill .@gid;
	return;
}

// F_Expedition_Reset() -> deja al personaje sin contrato: retira la marca y
// borra todo su estado. Se usa al entregar, abandonar o si el contrato ya no existe.
function	script	F_Expedition_Reset	{
	callfunc "F_Expedition_ClearMark";
	eb_qid = 0;
	eb_state = 0;
	eb_prog = 0;
	callfunc "F_Expedition_HuntEnd";
	return;
}

// F_Expedition_Spawn(id) -> 1 si la marca quedo puesta, 0 si no. Si la celda
// del contrato no es transitable, prueba celdas al azar cada vez mas lejos.
function	script	F_Expedition_Spawn	{
	.@id = getarg(0);
	if (!getvariableofnpc(.valid[.@id], "ExpeditionData"))
		return 0;

	.@map$ = getvariableofnpc(.map$[.@id], "ExpeditionData");
	.@x = getvariableofnpc(.x[.@id], "ExpeditionData");
	.@y = getvariableofnpc(.y[.@id], "ExpeditionData");
	.@mob = getvariableofnpc(.mob[.@id], "ExpeditionData");

	.@sx = .@x;
	.@sy = .@y;
	if (!checkcell(.@map$, .@sx, .@sy, cell_chkpass)) {
		.@sx = 0;
		for (.@r = 2; .@r <= 30 && .@sx == 0; .@r += 2) {
			for (.@t = 0; .@t < 24; .@t++) {
				.@nx = .@x - .@r + rand(0, .@r * 2);
				.@ny = .@y - .@r + rand(0, .@r * 2);
				if (checkcell(.@map$, .@nx, .@ny, cell_chkpass)) {
					.@sx = .@nx;
					.@sy = .@ny;
					break;
				}
			}
		}
		if (.@sx == 0) {
			debugmes "expedition-board: no walkable cell near " + .@map$ + " " + .@x + "," + .@y;
			return 0;
		}
	}

	callfunc "F_Expedition_ClearMark";
	monster .@map$, .@sx, .@sy, "Marca de expedicion", .@mob, 1, "ExpeditionCtrl::OnMarkDead";
	.@gid = $@mobid[0];
	if (.@gid < 1) {
		debugmes "expedition-board: monster() failed for contract " + .@id + " on " + .@map$;
		return 0;
	}
	$@eb_mark[getcharid(0)] = .@gid;
	$@eb_markowner[.@gid] = getcharid(0);
	return 1;
}

-	script	ExpeditionCtrl	-1,{
	end;

OnPCLoginEvent:
	eb_gid = 0;
	if (eb_state != 1)
		end;
	// Una caceria en curso vuelve a contar en $@eb_hunters (se vacia al reiniciar).
	if (eb_hunt_mob) {
		$@eb_hunters++;
		end;
	}
	// Solo las expediciones tienen marca que recolocar.
	if (getvariableofnpc(.type[eb_qid], "ExpeditionData") != 1)
		end;
	if (callfunc("F_Expedition_MarkAlive"))
		end;
	if (callfunc("F_Expedition_Spawn", eb_qid)) {
		callfunc "F_Expedition_Strengthen", eb_qid;
		dispbottom "Expedicion: la marca volvio a aparecer. El contrato sigue activo.";
	} else
		dispbottom "Expedicion: no se pudo recolocar la marca. Habla con el tablero.";
	end;

OnPCLogoutEvent:
	if (eb_hunt_mob) {
		if ($@eb_hunters > 0)
			$@eb_hunters--;
	}
	end;

// Corre con quien mato a la marca. Sin jugador (la marca se retiro por script)
// no hay nada que completar. Se acredita al dueno de la marca: al propio killer,
// o a su companero de grupo si esta conectado y en el mismo mapa.
OnMarkDead:
	if (!playerattached())
		end;
	.@gid = killedgid;
	.@owner = $@eb_markowner[.@gid];
	if (.@owner < 1)
		end;
	$@eb_markowner[.@gid] = 0;
	if (.@owner != getcharid(0)) {
		.@map$ = strcharinfo(3);
		.@pid = getcharid(1);
		if (!.@pid)
			end;
		.@aid = callfunc("F_Expedition_PartyAid", .@pid, .@owner);
		if (!.@aid)
			end;
		if (!attachrid(.@aid))
			end;
		if (strcharinfo(3) != .@map$)
			end;
	}
	// Desde aqui el personaje adjunto es el dueno del contrato.
	if (eb_state != 1 || $@eb_mark[getcharid(0)] != .@gid)
		end;
	$@eb_mark[getcharid(0)] = 0;
	eb_state = 2;
	dispbottom "Expedicion: objetivo abatido. Entrega con el Intendente de " + getvariableofnpc(.cityname$[eb_qid], "ExpeditionData") + ", junto al tablero.";
	// Entrada en el diario del viajero. Si adventure (su diario) no esta activo, doevent
	// solo deja una advertencia en el log del servidor.
	@tj_type$ = "note";
	@tj_text$ = "Marca abatida: " + getvariableofnpc(.name$[eb_qid], "ExpeditionData") + " (" + getvariableofnpc(.place$[eb_qid], "ExpeditionData") + ").";
	doevent "TravelJournal::OnLog";
	end;

// Corre con CADA monstruo que mata un jugador, en todo el servidor, bots del
// Population Engine incluidos. Por eso la primera linea es una sola
// comparacion: sin nadie cazando en el servidor ($@eb_hunters 0) termina sin
// mirar nada mas. Quien tiene caceria propia ya cuenta en $@eb_hunters, asi que
// esa salida no le quita ninguna muerte. Si alguien caza, cuenta la del que
// mata (eb_hunt_mob es 0 sin caceria, y se salta) y, si tiene grupo, se
// repasa a sus companeros conectados: cada uno cuenta si tiene caceria de esa
// especie y esta en el mismo mapa. Las lecturas de la tabla de contratos solo
// ocurren cuando una muerte cuenta.
OnNPCKillEvent:
	if ($@eb_hunters < 1)
		end;
	.@mob = killedrid;
	.@map$ = strcharinfo(3);
	if (eb_hunt_mob)
		callsub S_Hunt, .@mob, .@map$;
	.@pid = getcharid(1);
	if (!.@pid)
		end;
	.@self = getcharid(3);
	getpartymember .@pid, 2;
	.@n = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@aid[.@i] == .@self)
			continue;
		if (!attachrid(.@aid[.@i]))
			continue;
		if (eb_hunt_mob)
			callsub S_Hunt, .@mob, .@map$;
	}
	attachrid(.@self);
	end;

// S_Hunt(mob id, map): suma una muerte a la caceria del personaje adjunto si es
// de esa especie, en ese mapa y el esta en el. Avisa y completa al llegar al total.
// La especie es el nombre: cuenta cualquier id que se llame exactamente igual que
// eb_hunt_mob (Goblin 1122-1126), no uno de nombre parecido (Goblin Archer). Los
// nombres solo se leen en el mapa de la caceria y si el id no coincide.
S_Hunt:
	if (eb_state != 1 || strcharinfo(3) != eb_hunt_map$ || strcharinfo(3) != getarg(1))
		return;
	.@name$ = getmonsterinfo(eb_hunt_mob, MOB_NAME);
	if (getarg(0) != eb_hunt_mob) {
		if (getmonsterinfo(getarg(0), MOB_NAME) != .@name$)
			return;
	}
	eb_prog++;
	.@need = getvariableofnpc(.count[eb_qid], "ExpeditionData");
	if (eb_prog < .@need) {
		dispbottom "Caceria: " + eb_prog + "/" + .@need + " " + .@name$ + ".";
		return;
	}
	eb_state = 2;
	callfunc "F_Expedition_HuntEnd";
	dispbottom "Caceria completada: " + .@need + " " + .@name$ + ". Vuelve al tablero de " + getvariableofnpc(.cityname$[eb_qid], "ExpeditionData") + " a entregar.";
	@tj_type$ = "note";
	@tj_text$ = "Caceria completada: " + .@need + " " + .@name$ + " (" + getvariableofnpc(.place$[eb_qid], "ExpeditionData") + ").";
	doevent "TravelJournal::OnLog";
	return;
}

		mes .@hand$;
		.@tag$ = callfunc("F_Expedition_Tag", .@id);
		if (.@tag$ != "")
			mes .@tag$;
		mes "Recompensa: " + callfunc("F_Expedition_Zeny", .@id) + " zeny y " + getvariableofnpc(.amt[.@id], "ExpeditionData") + " " + getitemname(getvariableofnpc(.item[.@id], "ExpeditionData")) + ".";
		next;
		if (select("Aceptar el contrato:No por ahora") == 2) {
			mes "[Tablero de Expediciones]";
			mes "El aviso sigue clavado.";
			close;
		}
		mes "[Tablero de Expediciones]";
		if (.@type == 1) {
			if (!callfunc("F_Expedition_Spawn", .@id)) {
				mes "No se pudo colocar la marca. Prueba de nuevo o revisa el mapa.";
				close;
			}
		}
		eb_qid = .@id;
		eb_state = 1;
		eb_prog = 0;
		callfunc "F_Expedition_HuntEnd";
		// Una caceria se cuenta desde este momento, solo en su mapa.
		if (.@type == 2) {
			eb_hunt_mob = getvariableofnpc(.mob[.@id], "ExpeditionData");
			eb_hunt_map$ = getvariableofnpc(.map$[.@id], "ExpeditionData");
			$@eb_hunters++;
		}
		// Diario del viajero (ver OnMarkDead): si adventure no esta, solo hay un aviso en el log.
		@tj_type$ = "note";
		if (.@type == 2)
			@tj_text$ = "Acepte en " + getvariableofnpc(.cityname$[.@id], "ExpeditionData") + " cazar " + getvariableofnpc(.count[.@id], "ExpeditionData") + " " + getmonsterinfo(getvariableofnpc(.mob[.@id], "ExpeditionData"), MOB_NAME) + " en " + getvariableofnpc(.place$[.@id], "ExpeditionData") + ".";
		else
			@tj_text$ = "Acepte en " + getvariableofnpc(.cityname$[.@id], "ExpeditionData") + " el contrato: " + getvariableofnpc(.name$[.@id], "ExpeditionData") + " (" + getvariableofnpc(.place$[.@id], "ExpeditionData") + ").";
		doevent "TravelJournal::OnLog";
		if (.@type == 1) {
			mes "Contrato aceptado. Busca a ^0000FFMarca de expedicion^000000.";
			close2;
			// Despues de guardar el estado: si esto fallara, el contrato ya esta aceptado.
			callfunc "F_Expedition_Strengthen", .@id;
			end;
		}
		mes "Contrato aceptado. Cada pieza cuenta desde ahora, solo en ese mapa.";
		close;
	}
	close;
}
