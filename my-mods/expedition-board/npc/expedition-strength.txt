// expedition-board -- Marca mas fuerte segun el Tier del mapa.
//
// Las lineas de cabecera usan TAB entre campos, no espacios.
//
// Va en su propio archivo porque usa setunitdata con UMOB_MAXHP, UMOB_HP,
// UMOB_ATKMIN y UMOB_ATKMAX. Probado en el juego: la marca con refuerzo aguanta
// claramente mas que sin el. Si algun dia fallara, el error sale en el log del
// map-server al aceptar un contrato y la marca queda como un monstruo normal.
// Se apaga y se ajusta desde Settings -> Mods (mark_strength, mark_hp_tier,
// mark_atk_tier en mod.json).
//
// Se llama DESPUES de guardar el estado del contrato, nunca antes.

// F_Expedition_Strengthen(id) -> 1 si la marca del personaje se reforzo.
// Con el Tier del atlas T (1-6) y los valores por defecto:
//   HP    = 150% + 50% por cada Tier por encima del 1   (T1 x1.5 ... T6 x4)
//   ATK   = 100% + 15% por cada Tier por encima del 1   (T1 x1.0 ... T6 x1.75)
// Sin atlas (Tier 0) la marca no cambia. Una de elite, ademas, x2 de HP y
// x1,3 de ATK (elite_hp, elite_atk).
function	script	F_Expedition_Strengthen	{
	.@enabled = callfunc("F_ModSetting", "expedition-board", "mark_strength", 1);
	if (!.@enabled)
		return 0;

	.@tier = callfunc("F_Expedition_Tier", getarg(0));
	if (.@tier < 1)
		return 0;

	.@gid = $@eb_mark[getcharid(0)];
	if (.@gid < 1 || !unitexists(.@gid))
		return 0;

	getunitdata .@gid, .@u;
	.@hp_pct = 150 + callfunc("F_ModSetting", "expedition-board", "mark_hp_tier", 50) * (.@tier - 1);
	.@atk_pct = 100 + callfunc("F_ModSetting", "expedition-board", "mark_atk_tier", 15) * (.@tier - 1);
	// Una marca de elite (D40) suma elite_hp % de HP y elite_atk % de ATK encima.
	if (getvariableofnpc(.type[getarg(0)], "ExpeditionData") == 3) {
		.@hp_pct = .@hp_pct * (100 + callfunc("F_ModSetting", "expedition-board", "elite_hp", 100)) / 100;
		.@atk_pct = .@atk_pct * (100 + callfunc("F_ModSetting", "expedition-board", "elite_atk", 30)) / 100;
	}

	.@maxhp = .@u[UMOB_MAXHP] * .@hp_pct / 100;
	setunitdata .@gid, UMOB_MAXHP, .@maxhp;
	setunitdata .@gid, UMOB_HP, .@maxhp;
	setunitdata .@gid, UMOB_ATKMIN, .@u[UMOB_ATKMIN] * .@atk_pct / 100;
	setunitdata .@gid, UMOB_ATKMAX, .@u[UMOB_ATKMAX] * .@atk_pct / 100;
	return 1;
}
