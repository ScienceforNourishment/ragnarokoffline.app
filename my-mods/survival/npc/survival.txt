// F_Survival_IsSafe(map$) -> 1 when hunger and thirst do not drop on that map.
// Safe when the world-atlas mod, if loaded, says so, or when the map is in
// survival_mgr's own .safe_maps$ list. The atlas is a soft dependency: it sets
// $@wa_ready when its table is filled, and without it only the list applies.
function	script	F_Survival_IsSafe	{
	.@m$ = getarg(0);
	if ($@wa_ready) {
		if (callfunc("F_Atlas_IsSafe", .@m$))
			return 1;
	}
	return compare(getvariableofnpc(.safe_maps$, "survival_mgr"), "|" + .@m$ + "|");
}

// One clock for every player instead of an addtimer chain per player. A
// player's own events wait in a 2-slot queue while they talk to an NPC and the
// rest are dropped, which killed the old chains for the whole session.
// attachrid does not go through that queue. See CONVENTIONS.md, Temporizadores.
-	script	survival_mgr	-1,{
OnInit:
    // Maps where hunger and thirst do not drop. Main cities only; add an
    // interior or another settlement by appending its map name between bars.
    .safe_maps$ = "|prontera|izlude|geffen|payon|morocc|alberta|aldebaran|";
    // Testing rate: each need loses 1 point every 15 seconds.
    .need_every = 15;
    initnpctimer;
    end;

OnPCLoginEvent:
    set @clima_hunger_extra_loss, 0;
    set @clima_thirst_extra_loss, 0;
    set @clima_exposure_map$, "";
    if (hm_initialized == 0) {
    	set hm_hunger, 100;
    	set hm_initialized, 1;
    }
    if (thirst_initialized == 0) {
    	set thirst_value, 100;
    	set thirst_initialized, 1;
    }
    dispbottom "@@event hunger update " + hm_hunger;
    dispbottom "@@event thirst update " + thirst_value;
    @sv_need_at = gettimetick(2) + .need_every;
    callsub S_Join;
    end;

OnPCLogoutEvent:
    callsub S_Leave;
    set @hm_penalty_str, 0;
    set @hm_penalty_agi, 0;
    set @hm_penalty_vit, 0;
    set @hm_penalty_int, 0;
    set @hm_penalty_dex, 0;
    set @hm_penalty_luk, 0;
    set @thirst_penalty_str, 0;
    set @thirst_penalty_agi, 0;
    set @thirst_penalty_vit, 0;
    set @thirst_penalty_int, 0;
    set @thirst_penalty_dex, 0;
    set @thirst_penalty_luk, 0;
    set @clima_hunger_extra_loss, 0;
    set @clima_thirst_extra_loss, 0;
    set @clima_exposure_map$, "";
    end;

// Every 4 s, for each player in the list. The clock is restarted first, so a
// script error further down cannot stop it.
OnTimer4000:
    initnpctimer;
    for (.@i = .pl_n - 1; .@i >= 0; .@i--) {
        if (!attachrid(.pl[.@i])) {
            callsub S_Drop, .@i;
            continue;
        }
        callsub S_Tick;
        detachrid;
    }
    end;

S_Tick:
    .@now = gettimetick(2);
    if (.@now >= @sv_need_at) {
        @sv_need_at += .need_every;
        if (@sv_need_at <= .@now)
            @sv_need_at = .@now + .need_every;
        callsub S_Needs;
    }
    callsub S_Penalty;
    return;

S_Needs:
    .@map$ = strcharinfo(3);
    if (@clima_exposure_map$ != "" && @clima_exposure_map$ != .@map$) {
        set @clima_hunger_extra_loss, 0;
        set @clima_thirst_extra_loss, 0;
        set @clima_exposure_map$, "";
    }
    if (!callfunc("F_Survival_IsSafe", .@map$)) {
        if (hm_hunger > 0) {
            set hm_hunger, hm_hunger - 1;
            if (hm_hunger > 0 && @clima_hunger_extra_loss > 0 && @clima_exposure_map$ == .@map$)
                set hm_hunger, hm_hunger - 1;
        }
        if (thirst_value > 0) {
            set thirst_value, thirst_value - 1;
            if (thirst_value > 0 && @clima_thirst_extra_loss > 0 && @clima_exposure_map$ == .@map$)
                set thirst_value, thirst_value - 1;
        }
    }
    dispbottom "@@event hunger update " + hm_hunger;
    dispbottom "@@event thirst update " + thirst_value;
    return;

S_Penalty:
    if (hm_hunger == 0 || thirst_value == 0) {
    	// Restore this system's previous debuffs before recalculating both.
    	.@str = readparam(bStr) + @hm_penalty_str + @thirst_penalty_str;
    	.@agi = readparam(bAgi) + @hm_penalty_agi + @thirst_penalty_agi;
    	.@vit = readparam(bVit) + @hm_penalty_vit + @thirst_penalty_vit;
    	.@int = readparam(bInt) + @hm_penalty_int + @thirst_penalty_int;
    	.@dex = readparam(bDex) + @hm_penalty_dex + @thirst_penalty_dex;
    	.@luk = readparam(bLuk) + @hm_penalty_luk + @thirst_penalty_luk;

    	if (hm_hunger == 0) {
    		set @hm_penalty_str, .@str * 20 / 100;
    		set @hm_penalty_agi, .@agi * 20 / 100;
    		set @hm_penalty_vit, .@vit * 20 / 100;
    		set @hm_penalty_int, .@int * 20 / 100;
    		set @hm_penalty_dex, .@dex * 20 / 100;
    		set @hm_penalty_luk, .@luk * 20 / 100;
    	} else {
    		set @hm_penalty_str, 0;
    		set @hm_penalty_agi, 0;
    		set @hm_penalty_vit, 0;
    		set @hm_penalty_int, 0;
    		set @hm_penalty_dex, 0;
    		set @hm_penalty_luk, 0;
    	}

    	if (thirst_value == 0) {
    		set @thirst_penalty_str, .@str * 20 / 100;
    		set @thirst_penalty_agi, .@agi * 20 / 100;
    		set @thirst_penalty_vit, .@vit * 20 / 100;
    		set @thirst_penalty_int, .@int * 20 / 100;
    		set @thirst_penalty_dex, .@dex * 20 / 100;
    		set @thirst_penalty_luk, .@luk * 20 / 100;
    	} else {
    		set @thirst_penalty_str, 0;
    		set @thirst_penalty_agi, 0;
    		set @thirst_penalty_vit, 0;
    		set @thirst_penalty_int, 0;
    		set @thirst_penalty_dex, 0;
    		set @thirst_penalty_luk, 0;
    	}

    	.@penalty$ = "bonus bStr,-" + (@hm_penalty_str + @thirst_penalty_str) + ";";
    	.@penalty$ += "bonus bAgi,-" + (@hm_penalty_agi + @thirst_penalty_agi) + ";";
    	.@penalty$ += "bonus bVit,-" + (@hm_penalty_vit + @thirst_penalty_vit) + ";";
    	.@penalty$ += "bonus bInt,-" + (@hm_penalty_int + @thirst_penalty_int) + ";";
    	.@penalty$ += "bonus bDex,-" + (@hm_penalty_dex + @thirst_penalty_dex) + ";";
    	.@penalty$ += "bonus bLuk,-" + (@hm_penalty_luk + @thirst_penalty_luk) + ";";
    	.@speed_penalty = 0;
    	if (hm_hunger == 0)
    		set .@speed_penalty, .@speed_penalty + 20;
    	if (thirst_value == 0)
    		set .@speed_penalty, .@speed_penalty + 20;
    	.@penalty$ += "bonus bSpeedRate,-" + .@speed_penalty + ";";
    	bonus_script .@penalty$, 5, 1032;
    } else {
    	set @hm_penalty_str, 0;
    	set @hm_penalty_agi, 0;
    	set @hm_penalty_vit, 0;
    	set @hm_penalty_int, 0;
    	set @hm_penalty_dex, 0;
    	set @hm_penalty_luk, 0;
    	set @thirst_penalty_str, 0;
    	set @thirst_penalty_agi, 0;
    	set @thirst_penalty_vit, 0;
    	set @thirst_penalty_int, 0;
    	set @thirst_penalty_dex, 0;
    	set @thirst_penalty_luk, 0;
    }
    return;

// The clock's players, by account id. Population Engine bots never fire
// OnPCLoginEvent, so they never join.
S_Join:
    .@aid = getcharid(3);
    for (.@i = 0; .@i < .pl_n; .@i++)
        if (.pl[.@i] == .@aid)
            return;
    .pl[.pl_n] = .@aid;
    .pl_n++;
    return;

S_Leave:
    .@aid = getcharid(3);
    for (.@i = 0; .@i < .pl_n; .@i++) {
        if (.pl[.@i] == .@aid) {
            callsub S_Drop, .@i;
            return;
        }
    }
    return;

// Removes position getarg(0) by moving the last one into it.
S_Drop:
    .@i = getarg(0);
    .pl_n--;
    .pl[.@i] = .pl[.pl_n];
    .pl[.pl_n] = 0;
    return;
}
