//===== Ragnarok Offline: temperature =====================================
//= Temperatura del viajero. No es una barra que baje: se calcula cada pocos
//= segundos a partir de tres cosas y da un estado con nombre.
//=
//=   sensacion = temperatura del mapa (atlas, -2..2)
//=             + exposicion del clima (calor intenso +1, frio extremo -1)
//=             + abrigo de lo que llevas puesto (positivo abriga, negativo
//=               refresca)
//=
//=   -3 Congelado  -2 Helado  -1 Fresco  0 Comodo  1 Calido  2 Acalorado
//=   3 Sofocado.   +-1 es solo una sensacion, sin penalizacion.
//=
//= Premia ademas de castigar: si el lugar es frio o caluroso y tu ropa lo
//= compensa (sensacion 0), un bono pequeno de HP/SP maximos.
//= Nunca impide jugar: solo resta atributos o un poco de HP/SP maximos.
//=
//= En ciudades (F_Atlas_IsSafe) no hay temperatura. Sin world-atlas tampoco.
//= Los penalizadores usan su propio bonus_script (texto distinto del de
//= survival y wounds), asi que conviven con ellos. Un unico reloj del NPC
//= para todos los personajes (initnpctimer + attachrid), no un addtimer por
//= personaje: ver CONVENTIONS.md, Temporizadores.
//=
//= Dependencias: world-atlas (obligatoria). clima: opcional, solo se leen las
//= variables @clima_*_extra_loss y @clima_exposure_map$.
//===========================================================================

// F_Temperature_Gear() -> abrigo total del equipo puesto (positivo abriga).
function	script	F_Temperature_Gear	{
	setarray .@slot[0], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW, EQI_ARMOR, EQI_GARMENT, EQI_SHOES;
	.@n = getvariableofnpc(.g_count, "temp_mgr");
	.@total = 0;
	for (.@i = 0; .@i < getarraysize(.@slot); .@i++) {
		.@id = getequipid(.@slot[.@i]);
		if (.@id <= 0)
			continue;
		for (.@j = 0; .@j < .@n; .@j++) {
			if (getvariableofnpc(.g_id[.@j], "temp_mgr") == .@id) {
				.@total += getvariableofnpc(.g_val[.@j], "temp_mgr");
				break;
			}
		}
	}
	return .@total;
}

// F_Temperature_Feel() -> sensacion -3..3 del personaje en su mapa actual.
// Deja el desglose en @tp_env, @tp_clima y @tp_gear (0 si no hay temperatura).
function	script	F_Temperature_Feel	{
	@tp_env = 0;
	@tp_clima = 0;
	@tp_gear = 0;
	if (!$@wa_ready)
		return 0;
	.@map$ = strcharinfo(3);
	if (callfunc("F_Atlas_IsSafe", .@map$))
		return 0;
	@tp_env = callfunc("F_Atlas_Temp", .@map$);
	if (@clima_exposure_map$ == .@map$) {
		if (@clima_thirst_extra_loss)
			@tp_clima = 1;
		else if (@clima_hunger_extra_loss)
			@tp_clima = -1;
	}
	@tp_gear = callfunc("F_Temperature_Gear");
	.@feel = @tp_env + @tp_clima + @tp_gear;
	if (.@feel > 3)
		.@feel = 3;
	if (.@feel < -3)
		.@feel = -3;
	return .@feel;
}

// F_Temperature_Name(feel) -> nombre del estado.
function	script	F_Temperature_Name	{
	switch (getarg(0)) {
	case -3: return "Congelado";
	case -2: return "Helado";
	case -1: return "Fresco";
	case 1: return "Calido";
	case 2: return "Acalorado";
	case 3: return "Sofocado";
	}
	return "Comodo";
}

-	script	temp_mgr	-1,{
	end;

OnInit:
	// Abrigo de cada prenda: positivo abriga, negativo refresca. Los dos
	// primeros son del mod clima (si no esta, nunca coinciden con nada).
	.g_count = 6;
	.g_id[0] = 50833; .g_val[0] = 2;	// Abrigo Polar (clima)
	.g_id[1] = 50832; .g_val[1] = -1;	// Sombrero del Desierto (clima)
	.g_id[2] = 50840; .g_val[2] = 2;	// Capa de Piel
	.g_id[3] = 50841; .g_val[3] = 1;	// Chaleco de Lana
	.g_id[4] = 50842; .g_val[4] = -2;	// Manto de Lino
	.g_id[5] = 50843; .g_val[5] = -1;	// Tunica Fresca

	bindatcmd "temperatura", "temp_mgr::OnStatus", 0, 99;
	initnpctimer;
	end;

// @temperatura: estado actual y de donde sale.
OnStatus:
	.@feel = callfunc("F_Temperature_Feel");
	if (!$@wa_ready) {
		dispbottom "Temperatura: el atlas del mundo no esta activo.";
		end;
	}
	if (callfunc("F_Atlas_IsSafe", strcharinfo(3))) {
		dispbottom "Temperatura: en la ciudad estas a gusto.";
		end;
	}
	dispbottom "Temperatura: " + callfunc("F_Temperature_Name", .@feel) + " (" + .@feel + ").";
	dispbottom "Lugar " + @tp_env + ", clima " + @tp_clima + ", ropa " + @tp_gear + ".";
	end;

OnPCLoginEvent:
	@tp_last = 99;	// fuerza el aviso del primer estado
	callsub S_Join;
	end;

OnPCLogoutEvent:
	callsub S_Leave;
	end;

// Cada 4 s, para cada jugador de la lista. attachrid no pasa por la cola de
// eventos del personaje, asi que un dialogo abierto no detiene el reloj. Se
// rearma lo primero, para que un error de script mas abajo no lo pare.
OnTimer4000:
	initnpctimer;
	for (.@i = .pl_n - 1; .@i >= 0; .@i--) {
		if (!attachrid(.pl[.@i])) {
			callsub S_Drop, .@i;
			continue;
		}
		if (Hp > 0)
			callsub S_Apply;
		detachrid;
	}
	end;

// Duracion 5 s y renovacion cada 4 s, como wounds. Flag 1032 = reemplazar el
// identico + quitar al cerrar sesion. Al cambiar de estado, el penalizador
// anterior tarda hasta 5 s en agotarse.
S_Apply:
	.@feel = callfunc("F_Temperature_Feel");
	if (.@feel != @tp_last) {
		@tp_last = .@feel;
		switch (.@feel) {
		case -3: dispbottom "Estas congelado: el frio te cala hasta los huesos."; break;
		case -2: dispbottom "Estas helado: necesitas abrigo o volver a un sitio mas templado."; break;
		case -1: dispbottom "Sientes algo de fresco."; break;
		case 0: dispbottom "Te sientes comodo."; break;
		case 1: dispbottom "Sientes algo de calor."; break;
		case 2: dispbottom "Estas acalorado: necesitas ropa mas ligera o sombra."; break;
		case 3: dispbottom "Estas sofocado: el calor te agota."; break;
		}
	}
	if (.@feel == 0) {
		// Bien preparado: el lugar es extremo y tu ropa lo compensa.
		if (@tp_env + @tp_clima != 0)
			bonus_script "{ bonus bMaxHPrate,3; bonus bMaxSPrate,3; }", 5, 1032;
		return;
	}
	if (.@feel == 1 || .@feel == -1)
		return;
	if (.@feel == -2) {
		.@pts = 1 + BaseLevel / 20;
		if (.@pts > 6)
			.@pts = 6;
		bonus_script "{ bonus bAgi,-" + .@pts + "; }", 5, 1032, 0, EFST_COLD;
	} else if (.@feel == 2) {
		.@pts = 1 + BaseLevel / 20;
		if (.@pts > 6)
			.@pts = 6;
		bonus_script "{ bonus bInt,-" + .@pts + "; }", 5, 1032, 0, EFST_OVERHEAT;
	} else if (.@feel == -3) {
		.@pts = 2 + BaseLevel / 12;
		if (.@pts > 10)
			.@pts = 10;
		bonus_script "{ bonus bAgi,-" + .@pts + "; bonus bDex,-" + .@pts + "; bonus bMaxHPrate,-5; }", 5, 1032, 0, EFST_COLD;
	} else {
		.@pts = 2 + BaseLevel / 12;
		if (.@pts > 10)
			.@pts = 10;
		bonus_script "{ bonus bInt,-" + .@pts + "; bonus bVit,-" + .@pts + "; bonus bMaxSPrate,-5; }", 5, 1032, 0, EFST_OVERHEAT;
	}
	return;

// Jugadores del reloj, por account id. Los bots del Population Engine no
// disparan OnPCLoginEvent y nunca entran.
S_Join:
	.@aid = getcharid(3);
	for (.@i = 0; .@i < .pl_n; .@i++)
		if (.pl[.@i] == .@aid)
			return;
	.pl[.pl_n] = .@aid;
	.pl_n++;
	return;

S_Leave:
	.@aid = getcharid(3);
	for (.@i = 0; .@i < .pl_n; .@i++) {
		if (.pl[.@i] == .@aid) {
			callsub S_Drop, .@i;
			return;
		}
	}
	return;

// Quita la posicion getarg(0) poniendo en su lugar la ultima.
S_Drop:
	.@i = getarg(0);
	.pl_n--;
	.pl[.@i] = .pl[.pl_n];
	.pl[.pl_n] = 0;
	return;
}
