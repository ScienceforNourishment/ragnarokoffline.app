//===== Ragnarok Offline: temperature =====================================
//= Temperatura del viajero. No es una barra que baje: se calcula cada pocos
//= segundos a partir de tres cosas y da un estado con nombre.
//=
//=   sensacion = temperatura del mapa (atlas, -2..2)
//=             + exposicion del clima (calor intenso +1, frio extremo -1)
//=             + abrigo de lo que llevas puesto (positivo abriga, negativo
//=               refresca)
//=
//=   -3 Congelado  -2 Helado  -1 Fresco  0 Comodo  1 Calido  2 Acalorado
//=   3 Sofocado.   +-1 es solo una sensacion, sin penalizacion.
//=
//= Premia ademas de castigar: si el lugar es frio o caluroso y tu ropa lo
//= compensa (sensacion 0), un bono pequeno de HP/SP maximos.
//= Nunca impide jugar: solo resta atributos o un poco de HP/SP maximos.
//=
//= En ciudades (F_Atlas_IsSafe) no hay temperatura. Sin world-atlas tampoco.
//= Los penalizadores usan su propio bonus_script (texto distinto del de
//= survival y wounds), asi que conviven con ellos. Presion: publica @pr_temp
//= (0..3, la distancia a Comodo) y, por encima del tope comun (.pr_cap), reduce
//= su castigo en proporcion (BALANCE.md, Tope de presion). Un unico reloj del NPC
//= para todos los personajes (initnpctimer + attachrid), no un addtimer por
//= personaje: ver CONVENTIONS.md, Temporizadores.
//=
//= Dependencias: world-atlas (obligatoria). clima: opcional, solo se leen las
//= variables @clima_*_extra_loss y @clima_exposure_map$. campfire: opcional;
//= junto a una de sus fogatas (F_Campfire_Near) la sensacion es Comodo, sin
//= el bono de abrigo.
//===========================================================================

// F_Temperature_Gear() -> abrigo total del equipo puesto (positivo abriga).
function	script	F_Temperature_Gear	{
	setarray .@slot[0], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW, EQI_ARMOR, EQI_GARMENT, EQI_SHOES;
	.@n = getvariableofnpc(.g_count, "temp_mgr");
	.@total = 0;
	for (.@i = 0; .@i < getarraysize(.@slot); .@i++) {
		.@id = getequipid(.@slot[.@i]);
		if (.@id <= 0)
			continue;
		for (.@j = 0; .@j < .@n; .@j++) {
			if (getvariableofnpc(.g_id[.@j], "temp_mgr") == .@id) {
				.@total += getvariableofnpc(.g_val[.@j], "temp_mgr");
				break;
			}
		}
	}
	return .@total;
}

// F_Temperature_Feel() -> sensacion -3..3 del personaje en su mapa actual.
// Deja el desglose en @tp_env, @tp_clima y @tp_gear (0 si no hay temperatura),
// y @tp_fire a 1 junto a una fogata de campfire.
function	script	F_Temperature_Feel	{
	@tp_env = 0;
	@tp_clima = 0;
	@tp_gear = 0;
	@tp_fire = 0;
	if (!$@wa_ready)
		return 0;
	.@map$ = strcharinfo(3);
	if (callfunc("F_Atlas_IsSafe", .@map$))
		return 0;
	@tp_env = callfunc("F_Atlas_Temp", .@map$);
	if (@clima_exposure_map$ == .@map$) {
		if (@clima_thirst_extra_loss)
			@tp_clima = 1;
		else if (@clima_hunger_extra_loss)
			@tp_clima = -1;
	}
	@tp_gear = callfunc("F_Temperature_Gear");
	// Junto al fuego, Comodo. $@cf_live solo es distinto de 0 con campfire
	// cargado, asi que sin el no se llama a una funcion que no existe.
	if ($@cf_live) {
		if (callfunc("F_Campfire_Near")) {
			@tp_fire = 1;
			return 0;
		}
	}
	.@feel = @tp_env + @tp_clima + @tp_gear;
	if (.@feel > 3)
		.@feel = 3;
	if (.@feel < -3)
		.@feel = -3;
	// Rasgos de adventure (adv_trait_cold, adv_trait_heat): el frio o el calor
	// que castiga (2 o mas) cuenta un grado menos. Nunca da Comodo ni su bono.
	if (.@feel <= -2 && adv_trait_cold)
		.@feel++;
	else if (.@feel >= 2 && adv_trait_heat)
		.@feel--;
	return .@feel;
}

// F_Temperature_Name(feel) -> nombre del estado.
function	script	F_Temperature_Name	{
	switch (getarg(0)) {
	case -3: return "Congelado";
	case -2: return "Helado";
	case -1: return "Fresco";
	case 1: return "Calido";
	case 2: return "Acalorado";
	case 3: return "Sofocado";
	}
	return "Comodo";
}

-	script	temp_mgr	-1,{
	end;

OnInit:
	// Abrigo de cada prenda: positivo abriga, negativo refresca. Los dos
	// primeros son del mod clima (si no esta, nunca coinciden con nada).
	.g_count = 6;
	.g_id[0] = 50833; .g_val[0] = 2;	// Abrigo Polar (clima)
	.g_id[1] = 50832; .g_val[1] = -1;	// Sombrero del Desierto (clima)
	.g_id[2] = 50840; .g_val[2] = 2;	// Capa de Piel
	.g_id[3] = 50841; .g_val[3] = 1;	// Chaleco de Lana
	.g_id[4] = 50842; .g_val[4] = -2;	// Manto de Lino
	.g_id[5] = 50843; .g_val[5] = -1;	// Tunica Fresca

	// Tope de presion comun; survival y wounds usan el mismo numero.
	.pr_cap = 6;
	// Ajustes (mod.json). penalty_pct escala los castigos (100 = los de la
	// tabla del README); warm_bonus es el HP/SP maximo de Bien abrigado.
	.penalty_pct = callfunc("F_ModSetting", "temperature", "penalty_pct", 100);
	.warm_bonus = callfunc("F_ModSetting", "temperature", "warm_bonus", 3);
	bindatcmd "temperatura", "temp_mgr::OnStatus", 0, 99;
	initnpctimer;
	end;

// @temperatura: estado actual y de donde sale.
OnStatus:
	.@feel = callfunc("F_Temperature_Feel");
	if (!$@wa_ready) {
		dispbottom "Temperatura: el atlas del mundo no esta activo.";
		end;
	}
	if (callfunc("F_Atlas_IsSafe", strcharinfo(3))) {
		dispbottom "Temperatura: en la ciudad estas a gusto.";
		end;
	}
	dispbottom "Temperatura: " + callfunc("F_Temperature_Name", .@feel) + " (" + .@feel + ").";
	dispbottom "Lugar " + @tp_env + ", clima " + @tp_clima + ", ropa " + @tp_gear + ".";
	if (@tp_fire)
		dispbottom "Estas junto a una fogata: el fuego te mantiene a gusto.";
	end;

OnPCLoginEvent:
	@tp_last = 99;	// fuerza el aviso del primer estado
	callsub S_Join;
	end;

OnPCLogoutEvent:
	callsub S_Leave;
	end;

// El cliente termino de cargar el mapa (solo en mapas con loadevent): ya puede
// recibir el icono. Ver S_Apply.
OnPCLoadMapEvent:
	@tp_loaded$ = strcharinfo(3);
	end;

// Cada 4 s, para cada jugador de la lista. attachrid no pasa por la cola de
// eventos del personaje, asi que un dialogo abierto no detiene el reloj. Se
// rearma lo primero, para que un error de script mas abajo no lo pare.
OnTimer4000:
	initnpctimer;
	for (.@i = .pl_n - 1; .@i >= 0; .@i--) {
		if (!attachrid(.pl[.@i])) {
			callsub S_Drop, .@i;
			continue;
		}
		if (Hp > 0)
			callsub S_Apply;
		detachrid;
	}
	end;

// Duracion 5 s y renovacion cada 4 s, como wounds. Flag 1032 = reemplazar el
// identico + quitar al cerrar sesion. Al cambiar de estado, el penalizador
// anterior tarda hasta 5 s en agotarse.
S_Apply:
	// Primer tick en este mapa (o tras entrar al juego): ver el icono, abajo.
	.@arrived = (@tp_map$ != strcharinfo(3));
	@tp_map$ = strcharinfo(3);
	.@feel = callfunc("F_Temperature_Feel");
	if (.@feel < 0)
		@pr_temp = -.@feel;
	else
		@pr_temp = .@feel;
	if (.@feel != @tp_last) {
		@tp_last = .@feel;
		switch (.@feel) {
		case -3: dispbottom "Estas congelado: el frio te cala hasta los huesos."; break;
		case -2: dispbottom "Estas helado: necesitas abrigo o volver a un sitio mas templado."; break;
		case -1: dispbottom "Sientes algo de fresco."; break;
		case 0: dispbottom (@tp_fire ? "Junto al fuego entras en calor." : "Te sientes comodo."); break;
		case 1: dispbottom "Sientes algo de calor."; break;
		case 2: dispbottom "Estas acalorado: necesitas ropa mas ligera o sombra."; break;
		case 3: dispbottom "Estas sofocado: el calor te agota."; break;
		}
	}
	if (.@feel == 0) {
		// Bien preparado: el lugar es extremo y tu ropa lo compensa.
		if (@tp_env + @tp_clima != 0 && !@tp_fire)
			bonus_script "{ bonus bMaxHPrate," + .warm_bonus + "; bonus bMaxSPrate," + .warm_bonus + "; }", 5, 1032;
		// Comodo gracias a la ropa en un mapa de temperatura +-2: relato de
		// adventure (Media), una vez. @tp_told evita llamarlo en cada tick.
		if (!@tp_told && (@tp_env == 2 || @tp_env == -2) && !@tp_fire) {
			@tp_told = 1;
			if (getnpcid(0, "Adventure"))
				callfunc "F_Adv_Experience", "temp:extreme", 2, callfunc("F_Atlas_Tier", strcharinfo(3)), "Comodo en " + (@tp_env < 0 ? "pleno frio" : "pleno calor") + " gracias a mi ropa";
		}
		return;
	}
	if (.@feel == 1 || .@feel == -1)
		return;
	// El icono va en su propio bonus_script, de texto fijo por icono. Cada
	// bonus_script con icono lo enciende al crearse y lo APAGA al vencer, aunque
	// otro con el mismo icono siga activo; con el icono dentro de la penalizacion,
	// pasar de Helado a Congelado (o subir de nivel) cambiaba el texto, el viejo
	// vencia y apagaba el icono del nuevo. Ojo: EFST_COLD es tambien el icono de
	// Cristalizacion (SC_CRYSTALIZE) y el cliente lo nombra asi.
	// rAthena manda el icono una sola vez, al crear la entrada: con el flag 1024
	// (BSF_FORCE_REPLACE) renovarla solo alarga su duracion. Tras cambiar de mapa
	// ese paquete llegaba mientras el cliente cargaba, y roBrowser lo tira porque
	// su personaje aun no existe. Por eso el primer tick en un mapa no crea el
	// icono: la entrada del mapa anterior (renovada hace 4 s, dura 5) vence y lo
	// apaga. Despues se espera ademas a que el cliente termine de cargar el mapa
	// (OnPCLoadMapEvent apunta @tp_loaded$): un mapa lento de cargar tardaba mas
	// de 8 s y el icono se perdia igual. OnPCLoadMapEvent solo llega en mapas con
	// loadevent (adventure lo pone en los del atlas); sin el, se espera un tick.
	.@ready = (@tp_loaded$ == @tp_map$ || !getmapflag(@tp_map$, MF_LOADEVENT));
	if (!.@arrived && .@ready) {
		if (.@feel < 0)
			bonus_script "{ /* temperature: frio */ bonus bMaxHP,0; }", 5, 1032, 0, EFST_COLD;
		else
			bonus_script "{ /* temperature: calor */ bonus bMaxHP,0; }", 5, 1032, 0, EFST_OVERHEAT;
	}
	// Bajo el tope de presion, el castigo se reduce en proporcion (minimo 1).
	.@scale = callsub(S_CapScale) * .penalty_pct / 100;
	if (.@feel == -2 || .@feel == 2) {
		.@pts = 1 + BaseLevel / 20;
		if (.@pts > 6)
			.@pts = 6;
	} else {
		.@pts = 2 + BaseLevel / 12;
		if (.@pts > 10)
			.@pts = 10;
	}
	.@pts = .@pts * .@scale / 100;
	if (.@pts < 1)
		.@pts = 1;
	.@rate = 5 * .@scale / 100;
	if (.@feel == -2)
		bonus_script "{ bonus bAgi,-" + .@pts + "; }", 5, 1032;
	else if (.@feel == 2)
		bonus_script "{ bonus bInt,-" + .@pts + "; }", 5, 1032;
	else if (.@feel == -3)
		bonus_script "{ bonus bAgi,-" + .@pts + "; bonus bDex,-" + .@pts + "; bonus bMaxHPrate,-" + .@rate + "; }", 5, 1032;
	else
		bonus_script "{ bonus bInt,-" + .@pts + "; bonus bVit,-" + .@pts + "; bonus bMaxSPrate,-" + .@rate + "; }", 5, 1032;
	return;

// S_CapScale() -> % del castigo que se aplica bajo el tope de presion comun.
// La misma formula esta en survival y wounds.
S_CapScale:
	.@total = @pr_survival + @pr_temp + @pr_wounds;
	if (@pr_clima && @pr_clima_map$ == strcharinfo(3))
		.@total++;
	if (.@total <= .pr_cap)
		return 100;
	return .pr_cap * 100 / .@total;

// Jugadores del reloj, por account id. Los bots del Population Engine no
// disparan OnPCLoginEvent y nunca entran.
S_Join:
	.@aid = getcharid(3);
	for (.@i = 0; .@i < .pl_n; .@i++)
		if (.pl[.@i] == .@aid)
			return;
	.pl[.pl_n] = .@aid;
	.pl_n++;
	return;

S_Leave:
	.@aid = getcharid(3);
	for (.@i = 0; .@i < .pl_n; .@i++) {
		if (.pl[.@i] == .@aid) {
			callsub S_Drop, .@i;
			return;
		}
	}
	return;

// Quita la posicion getarg(0) poniendo en su lugar la ultima.
S_Drop:
	.@i = getarg(0);
	.pl_n--;
	.pl[.@i] = .pl[.pl_n];
	.pl[.pl_n] = 0;
	return;
}
