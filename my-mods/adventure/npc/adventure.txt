//===== Ragnarok Offline: adventure ========================================
//= Expeditions and experiences ("relatos"), and the traveller's journal that
//= used to be the travel-journal mod. Design: my-mods/ADVENTURE.md.
//=
//= An experience is something a character lives once: a new map, a new biome,
//= a first step into a Tier. It is not paid on the spot: it is kept as an untold
//= relato and paid as EXP when the character rests and tells it (the Cronista
//= in each city, or Adventure::OnRest from another mod). Telling closes the
//= expedition, which started on leaving a safe map.
//=
//= No timers. Map changes come from OnPCLoadMapEvent, which rAthena only fires
//= on maps with the loadevent mapflag; adventure-maps.txt (generated from the
//= atlas) sets it on every atlas map. Time outside is summed from gettimetick(2)
//= at each change.
//=
//= Mod store, scope "char" (everything the window reads is under "client"):
//=   x.<kind>.<id>     1 once that experience was had (x.map.<map> is also
//=                     "visited", for safe maps as well)
//=   client.n          how many journal entries were ever written
//=   client.e.<n>      "<unix>|<type>|<text>", the last 300 are kept
//=   client.p.<n>      untold relato: "<base>|<job>|<per mille>|<tier>|<text>"
//=   pk.<n>            its key in the store form ("map.prt_sewb4"), for adv_told$
//=
//= Character variables other mods read: adv_trait_* (rasgos) and, since 0.5.0,
//= adv_told$[] / adv_told_n (the keys of the relatos told; see F_Adv_Told).
//=   client.x.*        the expedition: on, no, start, out, since, deaths, top,
//=                     from, deep, pn (relatos written so far), streak (new
//=                     experiences in this expedition, for the racha)
//=   client.t.*        the journal's copy of the rasgos: cold, heat, forager,
//=                     tough (1 once had) and cold_s, heat_s, biomes, cures
//=                     (progress). The rasgos themselves are the character
//=                     variables adv_trait_*, see "Rasgos" below
//=
//= Other mods publish an experience with the player attached:
//=   if (getnpcid(0, "Adventure"))
//=       callfunc "F_Adv_Experience", "chest:" + .@map$, 1, .@tier, "Abri el cofre de ...", 1;
//= The last argument (optional) shares it with the party members online on the
//= same map, each gaining it once (the group rule in CONVENTIONS.md).
//= The store is per mod and belongs to the NPC that runs the script, so
//= F_Adv_Experience cannot write it from another mod: it queues the experience
//= in temporary character variables and fires Adventure::OnQueue. If that event
//= is dropped (a player's event queue has 2 slots), the experience waits and is
//= taken at the next map change or rest. Nothing is lost.
//=
//= The journal entry point of travel-journal still exists: TravelJournal::OnLog
//= with @tj_type$ and @tj_text$. Never run this mod and travel-journal together.
//===========================================================================

// F_Adv_Experience(key$, size, tier, text${, party}) -> 1 when queued. PUBLIC.
// key$: "map:prt_fild08", "wound:3"...; letters, digits, _ and one : or .
// size: 1 minor, 2 medium, 3 major, 4 milestone. tier: the atlas Tier of
// where it happened (0 = the character's own band). text$: a short sentence.
// party: 1 to share it with the party members online on this map.
function	script	F_Adv_Experience	{
	.@key$ = getarg(0);
	if (.@key$ == "" || getarg(3) == "")
		return 0;
	.@n = getarraysize(@adv_qk$);
	if (.@n >= 32)
		return 0;
	@adv_qk$[.@n] = .@key$;
	@adv_qs[.@n] = getarg(1);
	@adv_qt[.@n] = getarg(2);
	@adv_qx$[.@n] = getarg(3);
	// The map is kept so the party is looked for where it happened, not where
	// the character is when the queue is taken.
	@adv_qm$[.@n] = (getarg(4, 0) ? strcharinfo(3) : "");
	doevent "Adventure::OnQueue";
	return 1;
}

// F_Adv_Note(type$, text$) -> 1 when queued. PUBLIC: a journal entry from
// another mod, with the player attached. Like F_Adv_Experience it only queues
// (@adv_nt$, @adv_nx$) and fires Adventure::OnQueue, so a dropped event loses
// nothing: the note is written at the next map change, rest or @relatos.
// Prefer it to doevent "TravelJournal::OnLog", which is lost when dropped.
function	script	F_Adv_Note	{
	if (getarg(0) == "" || getarg(1) == "")
		return 0;
	.@n = getarraysize(@adv_nt$);
	if (.@n >= 32)
		return 0;
	@adv_nt$[.@n] = getarg(0);
	@adv_nx$[.@n] = getarg(1);
	doevent "Adventure::OnQueue";
	return 1;
}

// The functions below use the mod store, which belongs to the mod of the NPC
// running the script: call them only from this mod's NPCs.

// F_Adv_Write(type$, text$): appends one journal entry.
function	script	F_Adv_Write	{
	.@type$ = replacestr(getarg(0), "|", "/");
	.@text$ = replacestr(getarg(1), "|", "/");
	if (.@type$ == "" || .@text$ == "")
		return;
	.@n = modstore_inc("char", "client.n", 1);
	modstore_set "char", "client.e." + .@n, gettimetick(2) + "|" + .@type$ + "|" + .@text$;
	if (.@n > 300)
		modstore_delete "char", "client.e." + (.@n - 300);
	dispbottom "@@event journal new";
	return;
}

// F_Adv_Place(map$) -> the atlas name of a map, or its code.
function	script	F_Adv_Place	{
	.@code$ = getarg(0);
	if ($@wa_ready) {
		.@name$ = callfunc("F_Atlas_Name", .@code$);
		if (.@name$ != "")
			return .@name$;
	}
	return .@code$;
}

// F_Adv_Exp(next, size, tier, streak) -> EXP of one relato for the attached
// character, from NextBaseExp or NextJobExp (ADVENTURE.md, "Cuanta EXP da una
// experiencia"). streak: the % the racha adds, 0 to 50.
function	script	F_Adv_Exp	{
	.@next = getarg(0);
	.@size = getarg(1);
	.@tier = getarg(2);
	.@streak = getarg(3, 0);
	if (.@next <= 0)
		return 0;
	// % of a level by size: minor, medium, major, milestone (settings, read in
	// the Adventure NPC's OnInit).
	if (.@size < 1)
		.@size = 1;
	if (.@size > 4)
		.@size = 4;
	// The character's band, from the dungeon-chest table.
	.@band = 1;
	if (BaseLevel >= 150) .@band = 6;
	else if (BaseLevel >= 120) .@band = 5;
	else if (BaseLevel >= 90) .@band = 4;
	else if (BaseLevel >= 60) .@band = 3;
	else if (BaseLevel >= 40) .@band = 2;
	// Tier factor x10: what is below you stops paying, above you pays double.
	.@f = 10;
	if (.@tier > 0) {
		.@d = .@tier - .@band;
		if (.@d <= -2) .@f = 1;
		else if (.@d == -1) .@f = 4;
		else if (.@d == 1) .@f = 15;
		else if (.@d >= 2) .@f = 20;
	}
	.@e = .@next * getvariableofnpc(.pct[.@size], "Adventure") * .@f / 1000;
	.@e = .@e * (100 + .@streak) / 100;
	// One relato never pays more than .cap % of a level (60), racha included.
	.@cap = .@next * getvariableofnpc(.cap, "Adventure") / 100;
	if (.@e > .@cap)
		.@e = .@cap;
	if (.@e < 1)
		.@e = 1;
	return .@e;
}

// F_Adv_Gain(key$, size, tier, text$) -> 1 when it is a new experience; it is
// then kept as an untold relato, with its EXP worked out now.
function	script	F_Adv_Gain	{
	.@k$ = "x." + strtolower(replacestr(getarg(0), ":", "."));
	if (modstore_get("char", .@k$, 0))
		return 0;
	modstore_set "char", .@k$, 1;
	// A key the store refuses (bad characters, too deep) is not written.
	if (modstore_get("char", .@k$, 0) != 1) {
		debugmes "adventure: experience key refused: " + .@k$;
		return 0;
	}
	.@size = getarg(1);
	.@tier = getarg(2);
	.@text$ = replacestr(getarg(3), "|", "/");
	// Racha: +.streak_step % (10) for each new experience already had in this
	// expedition, up to +.streak_max % (50). The first one pays plain.
	.@streak = modstore_get("char", "client.x.streak", 0);
	modstore_inc "char", "client.x.streak", 1;
	.@bonus = min(.@streak * getvariableofnpc(.streak_step, "Adventure"), getvariableofnpc(.streak_max, "Adventure"));
	.@base = callfunc("F_Adv_Exp", NextBaseExp, .@size, .@tier, .@bonus);
	.@job = callfunc("F_Adv_Exp", NextJobExp, .@size, .@tier, .@bonus);
	.@pm = 0;
	if (NextBaseExp > 0)
		.@pm = .@base * 1000 / NextBaseExp;
	.@n = modstore_inc("char", "client.x.pn", 1);
	modstore_set "char", "client.p." + .@n, .@base + "|" + .@job + "|" + .@pm + "|" + .@tier + "|" + .@text$;
	// Its key, for adv_told$ when it is told (outside "client": the window
	// does not need it).
	modstore_set "char", "pk." + .@n, strtolower(replacestr(getarg(0), ":", "."));
	// A Mayor or Hito relato may be an achievement, completed when told.
	if (.@size >= 3)
		callfunc "F_Adv_AchQueue", strtolower(replacestr(getarg(0), ":", ".")), .@size;
	callfunc "F_Adv_Write", "exp", .@text$;
	dispbottom "Nuevo relato: " + .@text$ + ".";
	return 1;
}

//--- Achievements (D39): Mayor and Hito relatos, completed when they are told,
//--- like their EXP. Ids 901000-901999; renewal/db/achievement_db.yml and
//--- System/achievement_list.lub. The waiting ones are in the store, "ach.<id>".
//=   tier.<n>       901000 + n     Hito, first step into Tier n
//=   chesttier.<n>  901010 + n     first chest of Tier n (dungeon-chest)
//=   mvp.*          901021         the first MVP
//=   nemesis.*      901031         the first nemesis; a Hito (rank 3) also 901032
//=   refuge.*       901041         the first refuge restored (refuges)

// F_Adv_AchQueue(key, size): keeps the achievements of a new relato until it is
// told. key: the store form, "tier.3", "mvp.1038".
function	script	F_Adv_AchQueue	{
	explode(.@k$, getarg(0), ".");
	.@kind$ = .@k$[0];
	.@v = atoi(.@k$[1]);
	if (.@kind$ == "tier" && .@v >= 1 && .@v <= 6)
		.@id[0] = 901000 + .@v;
	else if (.@kind$ == "chesttier" && .@v >= 1 && .@v <= 6)
		.@id[0] = 901010 + .@v;
	else if (.@kind$ == "mvp")
		.@id[0] = 901021;
	else if (.@kind$ == "nemesis") {
		.@id[0] = 901031;
		if (getarg(1) >= 4)
			.@id[1] = 901032;
	} else if (.@kind$ == "refuge")
		.@id[0] = 901041;
	for (.@i = 0; .@i < getarraysize(.@id); .@i++)
		modstore_set "char", "ach." + .@id[.@i], 1;
	return;
}

// F_Adv_AchTell: completes the achievements of the relatos just told. Only
// Renewal has them (D1; several rewards are not in Pre-Renewal's tables).
function	script	F_Adv_AchTell	{
	.@n = modstore_keys("char", "ach", .@ids$);
	if (!.@n)
		return;
	modstore_delete "char", "ach";
	if (!checkre(0))
		return;
	for (.@i = 0; .@i < .@n; .@i++) {
		.@id = atoi(.@ids$[.@i]);
		if (!achievementexists(.@id))
			achievementcomplete .@id;
	}
	return;
}

// F_Adv_Drain({share}): writes the notes queued with F_Adv_Note and takes the
// experiences queued with F_Adv_Experience. share 1 also gives the party ones to
// the party (F_Adv_Share); only from an event, never inside a dialog, because it
// attaches other players. Without it a party experience is the character's own.
function	script	F_Adv_Drain	{
	.@nn = getarraysize(@adv_nt$);
	if (.@nn) {
		copyarray .@nt$[0], @adv_nt$[0], .@nn;
		copyarray .@nx$[0], @adv_nx$[0], .@nn;
		deletearray @adv_nt$;
		deletearray @adv_nx$;
		for (.@i = 0; .@i < .@nn; .@i++)
			callfunc "F_Adv_Write", .@nt$[.@i], .@nx$[.@i];
	}
	.@n = getarraysize(@adv_qk$);
	if (.@n == 0)
		return;
	// Copied first: F_Adv_Gain could run another script that queues more.
	copyarray .@k$[0], @adv_qk$[0], .@n;
	copyarray .@s[0], @adv_qs[0], .@n;
	copyarray .@t[0], @adv_qt[0], .@n;
	copyarray .@x$[0], @adv_qx$[0], .@n;
	copyarray .@m$[0], @adv_qm$[0], .@n;
	deletearray @adv_qk$;
	deletearray @adv_qs;
	deletearray @adv_qt;
	deletearray @adv_qx$;
	deletearray @adv_qm$;
	for (.@i = 0; .@i < .@n; .@i++) {
		callfunc "F_Adv_Gain", .@k$[.@i], .@s[.@i], .@t[.@i], .@x$[.@i];
		if (getarg(0, 0) && .@m$[.@i] != "")
			callfunc "F_Adv_Share", .@k$[.@i], .@s[.@i], .@t[.@i], .@x$[.@i], .@m$[.@i];
	}
	return;
}

// F_Adv_Share(key$, size, tier, text$, map$): the same experience for each party
// member online on map$ (CONVENTIONS.md, "Regla comun de grupo"). Each one
// gains it once, with their own level and racha. Ends attached to the caller.
function	script	F_Adv_Share	{
	.@pid = getcharid(1);
	if (!.@pid)
		return;
	.@self = getcharid(3);
	getpartymember .@pid, 2;
	.@n = $@partymembercount;
	copyarray .@aid[0], $@partymemberaid[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@aid[.@i] == .@self)
			continue;
		if (!attachrid(.@aid[.@i]))
			continue;
		if (strcharinfo(3) == getarg(4))
			callfunc "F_Adv_Gain", getarg(0), getarg(1), getarg(2), getarg(3);
	}
	attachrid(.@self);
	return;
}

// F_Adv_CloseSegment: adds the time since the character left a safe map.
function	script	F_Adv_CloseSegment	{
	.@since = modstore_get("char", "client.x.since", 0);
	if (.@since <= 0)
		return;
	.@d = gettimetick(2) - .@since;
	if (.@d > 0)
		modstore_inc "char", "client.x.out", .@d;
	modstore_set "char", "client.x.since", 0;
	return;
}

// F_Adv_Duration(seconds) -> "1 h 42 min", "17 min".
function	script	F_Adv_Duration	{
	.@m = getarg(0) / 60;
	if (.@m < 60)
		return .@m + " min";
	return (.@m / 60) + " h " + (.@m % 60) + " min";
}

// F_Adv_Told(key$): publishes a relato just told in the character variables
// adv_told$[] (a ring of the last 64 keys, in the store form "map.prt_sewb4")
// and adv_told_n (how many were ever published). Other mods read them to react
// to what was told (tavern-rumors, RUMORES.md): entry k is adv_told$[k % 64],
// for k from max(seen, adv_told_n - 64) to adv_told_n - 1. Relatos gained
// before 0.5.0 have no key and are not published.
function	script	F_Adv_Told	{
	if (getarg(0) == "")
		return;
	adv_told$[adv_told_n % 64] = getarg(0);
	adv_told_n++;
	return;
}

// F_Adv_Rest -> how many relatos were told; -1 when there was nothing to tell
// and no expedition to close. Pays them one by one, in order, writes the
// expedition summary and ends the expedition.
function	script	F_Adv_Rest	{
	callfunc "F_Adv_Drain";
	callfunc "F_Adv_CloseSegment";
	.@on = modstore_get("char", "client.x.on", 0);
	.@pn = modstore_get("char", "client.x.pn", 0);
	if (!.@on && .@pn == 0)
		return -1;
	.@told = 0;
	.@pm = 0;
	for (.@i = 1; .@i <= .@pn; .@i++) {
		.@r$ = modstore_get("char", "client.p." + .@i, "");
		if (.@r$ == "")
			continue;
		explode(.@f$, .@r$, "|");
		.@b = atoi(.@f$[0]);
		.@j = atoi(.@f$[1]);
		// getexp refuses 0 and 0 (a character at the top level of both).
		if (.@b > 0 || .@j > 0)
			getexp .@b, .@j;
		.@pm += atoi(.@f$[2]);
		.@told++;
		callfunc "F_Adv_Told", modstore_get("char", "pk." + .@i, "");
	}
	modstore_delete "char", "client.p";
	modstore_delete "char", "pk";
	modstore_set "char", "client.x.pn", 0;
	modstore_set "char", "client.x.streak", 0;
	callfunc "F_Adv_AchTell";
	if (.@on) {
		.@no = modstore_get("char", "client.x.no", 0);
		.@out = modstore_get("char", "client.x.out", 0);
		.@from$ = modstore_get("char", "client.x.from", "");
		.@deep$ = modstore_get("char", "client.x.deep", "");
		.@deaths = modstore_get("char", "client.x.deaths", 0);
		.@s$ = "Expedición " + .@no + " · " + callfunc("F_Adv_Duration", .@out) + " fuera · " + .@from$;
		if (.@deep$ != "" && .@deep$ != .@from$)
			.@s$ += " → " + .@deep$;
		if (.@told == 1)
			.@s$ += " · 1 relato";
		else
			.@s$ += " · " + .@told + " relatos";
		if (.@told)
			.@s$ += " (+" + (.@pm / 10) + " % de nivel)";
		if (.@deaths == 1)
			.@s$ += " · 1 caída";
		else if (.@deaths > 1)
			.@s$ += " · " + .@deaths + " caídas";
		callfunc "F_Adv_Write", "expedition", .@s$;
		modstore_set "char", "client.x.on", 0;
		modstore_set "char", "client.x.out", 0;
		modstore_set "char", "client.x.deaths", 0;
	}
	return .@told;
}

// F_Adv_BiomeName(biome$) -> "la nieve", "las ruinas"...
function	script	F_Adv_BiomeName	{
	.@b$ = getarg(0);
	if (.@b$ == "bosque") return "el bosque";
	if (.@b$ == "costa") return "la costa";
	if (.@b$ == "cueva") return "las cuevas";
	if (.@b$ == "desierto") return "el desierto";
	if (.@b$ == "montana") return "la montaña";
	if (.@b$ == "nieve") return "la nieve";
	if (.@b$ == "pantano") return "el pantano";
	if (.@b$ == "pradera") return "la pradera";
	if (.@b$ == "ruinas") return "las ruinas";
	if (.@b$ == "selva") return "la selva";
	if (.@b$ == "volcan") return "tierra volcánica";
	return .@b$;
}

//--- Rasgos (ADVENTURE.md, "Rasgos"): permanent character variables that the
//--- mod owning each system reads to soften its own penalty. Character
//--- variables are seen from every NPC, so the readers need neither the store
//--- nor requires.mods. A trait reduces a pressure, it never cancels it.
//=   adv_trait_cold     frio cuenta 1 menos (temperature)
//=   adv_trait_heat     calor cuenta 1 menos (temperature)
//=   adv_trait_tough    las heridas restan 1 punto menos (wounds)
//=   adv_trait_forager  hambre y sed bajan un 20 % mas lento (survival)
//= Their progress, also character variables: adv_t_cold and adv_t_heat
//= (seconds on snow, or on desert and volcano maps) and adv_cured (one bit per
//= wound id ever cured). The journal reads a copy in the store (client.t.*).

// F_Adv_Cured(id): a wound or curse was cured. PUBLIC; wounds calls it after a
// real cure (not the GM command). Only character variables, so it works from
// any mod's NPC. Grants adv_trait_tough at the .tough_cures-th distinct one.
function	script	F_Adv_Cured	{
	.@id = getarg(0);
	if (.@id < 1 || .@id > 30)
		return;
	adv_cured = adv_cured | (1 << .@id);
	if (adv_trait_tough)
		return;
	if (callfunc("F_Adv_Bits", adv_cured) < getvariableofnpc(.tough_cures, "Adventure"))
		return;
	adv_trait_tough = 1;
	dispbottom "Nuevo rasgo: Curtido por las heridas. Las heridas te restan menos.";
	return;
}

// F_Adv_Bits(n) -> how many bits are set in n (bits 1 to 30).
function	script	F_Adv_Bits	{
	.@n = getarg(0);
	.@c = 0;
	for (.@i = 1; .@i <= 30; .@i++)
		if (.@n & (1 << .@i))
			.@c++;
	return .@c;
}

// F_Adv_TraitClock(kind$): adds the time on the last cold or hot map to its
// counter and starts timing the new one. kind$: "cold", "heat" or "" (any
// other map, a city, logging out).
function	script	F_Adv_TraitClock	{
	.@now = gettimetick(2);
	if (@adv_tat > 0) {
		.@d = .@now - @adv_tat;
		if (.@d > 0) {
			if (@adv_tk$ == "cold")
				adv_t_cold += .@d;
			else if (@adv_tk$ == "heat")
				adv_t_heat += .@d;
		}
	}
	@adv_tk$ = getarg(0);
	@adv_tat = (@adv_tk$ != "" ? .@now : 0);
	return;
}

// F_Adv_Biomes() -> how many biomes the character has a relato of.
function	script	F_Adv_Biomes	{
	setarray .@b$[0], "bosque", "costa", "cueva", "desierto", "montana", "nieve", "pantano", "pradera", "ruinas", "selva", "volcan";
	.@c = 0;
	for (.@i = 0; .@i < getarraysize(.@b$); .@i++)
		if (modstore_get("char", "x.biome." + .@b$[.@i], 0))
			.@c++;
	return .@c;
}

// F_Adv_Traits: grants the traits whose condition is met, writes each new one
// in the journal and refreshes the journal's copy of the progress. Store use:
// only from this mod's NPCs.
function	script	F_Adv_Traits	{
	.@secs = getvariableofnpc(.trait_minutes, "Adventure") * 60;
	.@biomes = callfunc("F_Adv_Biomes");
	.@cures = callfunc("F_Adv_Bits", adv_cured);
	if (!adv_trait_cold && adv_t_cold >= .@secs) {
		adv_trait_cold = 1;
		dispbottom "Nuevo rasgo: Hecho al frío. El frío te pesa menos.";
	}
	if (!adv_trait_heat && adv_t_heat >= .@secs) {
		adv_trait_heat = 1;
		dispbottom "Nuevo rasgo: Hecho al calor. El calor te pesa menos.";
	}
	if (!adv_trait_forager && .@biomes >= getvariableofnpc(.forager_biomes, "Adventure")) {
		adv_trait_forager = 1;
		dispbottom "Nuevo rasgo: Buscavidas. El hambre y la sed tardan más en llegar.";
	}
	if (!adv_trait_tough && .@cures >= getvariableofnpc(.tough_cures, "Adventure")) {
		adv_trait_tough = 1;
		dispbottom "Nuevo rasgo: Curtido por las heridas. Las heridas te restan menos.";
	}
	// The journal entry, once per trait, also for one F_Adv_Cured granted from
	// another mod (which cannot write this store).
	setarray .@k$[0], "cold", "heat", "forager", "tough";
	setarray .@t$[0], "Rasgo: Hecho al frío", "Rasgo: Hecho al calor", "Rasgo: Buscavidas", "Rasgo: Curtido por las heridas";
	for (.@i = 0; .@i < 4; .@i++) {
		if (getd("adv_trait_" + .@k$[.@i]) && !modstore_get("char", "client.t." + .@k$[.@i], 0)) {
			modstore_set "char", "client.t." + .@k$[.@i], 1;
			callfunc "F_Adv_Write", "trait", .@t$[.@i];
		}
	}
	// Progress for the Rasgos tab, written only when it changed.
	setarray .@pk$[0], "cold_s", "heat_s", "biomes", "cures";
	setarray .@pv[0], adv_t_cold, adv_t_heat, .@biomes, .@cures;
	for (.@i = 0; .@i < 4; .@i++)
		if (modstore_get("char", "client.t." + .@pk$[.@i], 0) != .@pv[.@i])
			modstore_set "char", "client.t." + .@pk$[.@i], .@pv[.@i];
	return;
}

// F_Adv_Long(seconds out): the relatos of a long expedition, 30, 60 and 120
// minutes outside (Media), each once per character. Tier: the deepest one.
function	script	F_Adv_Long	{
	.@out = getarg(0);
	if (.@out < 1800)
		return;
	.@top = modstore_get("char", "client.x.top", 0);
	callfunc "F_Adv_Gain", "long.30", 2, .@top, "Más de media hora fuera de las ciudades";
	if (.@out >= 3600)
		callfunc "F_Adv_Gain", "long.60", 2, .@top, "Más de una hora fuera de las ciudades";
	if (.@out >= 7200)
		callfunc "F_Adv_Gain", "long.120", 2, .@top, "Más de dos horas fuera de las ciudades";
	return;
}

// F_Adv_Close(): the "por los pelos" relato (Media, once) on reaching a safe
// map from outside with under 20 % HP, or with hunger or thirst at 0 (survival,
// soft: its variables only count once survival set them up).
function	script	F_Adv_Close	{
	.@close = (MaxHp > 0 && Hp * 100 < MaxHp * 20);
	if (hm_initialized && hm_hunger <= 0)
		.@close = 1;
	if (thirst_initialized && thirst_value <= 0)
		.@close = 1;
	if (.@close)
		callfunc "F_Adv_Gain", "close.return", 2, modstore_get("char", "client.x.top", 0), "Volví por los pelos";
	return;
}

// Runs with EVERY monster a player kills, Population Engine bots included
// (CONVENTIONS.md, "Muertes de monstruos"), so the first line drops all that
// is not an MVP (MD_MVP, as nemesis checks). Then only real players (@adv_on).
// The first MVP of each species is a relato (Mayor), shared with the party
// on the map.
OnNPCKillEvent:
	if (!(getmonsterinfo(killedrid, MOB_MODE) & MD_MVP))
		end;

		end;
	.@mob = killedrid;

	.@tier = 0;
	if ($@wa_ready)
		.@tier = callfunc("F_Atlas_Tier", .@map$);
	.@text$ = "Abatí a " + getmonsterinfo(.@mob, MOB_NAME);
	callfunc "F_Adv_Gain", "mvp." + .@mob, 3, .@tier, .@text$;
	callfunc "F_Adv_Share", "mvp." + .@mob, 3, .@tier, .@text$, .@map$;
	end;

OnQueue:

	end;

// PUBLIC, the form ADVENTURE.md first described: @adv_key$, @adv_size,
// @adv_tier, @adv_text$ and @adv_party (1 to share it with the party on this
// map). Prefer F_Adv_Experience: if this event is dropped, these variables are
// overwritten by the next caller.
OnExperience:
	if (@adv_key$ != "" && @adv_text$ != "") {
		.@n = getarraysize(@adv_qk$);
		@adv_qk$[.@n] = @adv_key$;
		@adv_qs[.@n] = @adv_size;
		@adv_qt[.@n] = @adv_tier;
		@adv_qx$[.@n] = @adv_text$;
		@adv_qm$[.@n] = (@adv_party ? strcharinfo(3) : "");
	}
	@adv_key$ = "";
	@adv_size = 0;
	@adv_tier = 0;
	@adv_text$ = "";
	@adv_party = 0;
	callfunc "F_Adv_Drain", 1;
	end;

	if (.@told > 0)
		dispbottom "Cuentas " + .@told + " relato" + (.@told > 1 ? "s" : "") + ".";
	end;
