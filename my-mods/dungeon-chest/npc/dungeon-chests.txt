function	script	F_DungeonChest_Move	{
	.@npc$ = getarg(0);
	.@map$ = getarg(1);
	.@placed = 0;
	for (.@try = 0; .@try < 500; .@try++) {
		.@x = rand(20, 280);
		.@y = rand(20, 280);
		if (!checkcell(.@map$, .@x, .@y, cell_chkpass))
			continue;
		movenpc .@npc$, .@x, .@y;
		.@placed = 1;
		break;
	}

	if (!.@placed)
		debugmes "dungeon-chest: could not find a passable cell on " + .@map$ + " after 500 attempts for " + .@npc$;
	return .@placed;
}

function	script	F_DungeonChest_Open	{
	.@npc$ = getarg(0);
	.@tier = getarg(1);
	.@item = getarg(2);

	mes "[Cofre de Tier " + .@tier + "]";
	if (getvariableofnpc(.opened, .@npc$)) {
		mes "El cofre ya fue abierto. No queda botín.";
		return 0;
	}

	mes "El cofre está cerrado. ¿Quieres abrirlo?";
	next;
	if (select("Abrirlo:Dejarlo cerrado") == 2)
		return 0;

	if (getvariableofnpc(.opened, .@npc$)) {
		mes "[Cofre de Tier " + .@tier + "]";
		mes "Alguien más llegó primero y ya abrió el cofre.";
		return 0;
	}

	if (!checkweight(.@item, 1)) {
		mes "[Cofre de Tier " + .@tier + "]";
		mes "No tienes espacio suficiente para cargar el botín.";
		return 0;
	}

	set getvariableofnpc(.opened, .@npc$), 1;
	getitem .@item, 1;
	mes "[Cofre de Tier " + .@tier + "]";
	mes "Encuentras " + getitemname(.@item) + ".";
	return 1;
}

function	script	F_DungeonChest_Tick	{
	.@npc$ = getarg(0);
	.@map$ = getarg(1);

	if (getvariableofnpc(.opened, .@npc$)) {
		set getvariableofnpc(.opened, .@npc$), 0;
		set getvariableofnpc(.relocated, .@npc$), 0;
		callfunc("F_DungeonChest_Move", .@npc$, .@map$);
	} else if (!getvariableofnpc(.relocated, .@npc$)) {
		callfunc("F_DungeonChest_Move", .@npc$, .@map$);
		set getvariableofnpc(.relocated, .@npc$), 1;
	}
	return;
}
