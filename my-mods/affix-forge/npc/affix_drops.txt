// Affix Forge: affix drops
//
// Every kill, each piece of equipment in the monster's own drop list gets an
// extra roll at its normal drop rate (times the "drop_mult" setting). A win
// hands the player a copy with 1-4 random bonus attributes (item random
// options) straight into the inventory.
//
// The server cannot alter the stock ground drop, so that one still falls
// plain and independently of this roll.

-	script	AffixForge_Drops	-1,{
	end;

OnInit:

	.pillar  = callfunc("F_ModSetting", "affix-forge", "ground_pillar", 1);
	if (!.ground) .pillar = 0; // the pillar only exists with ground drops
	.showmsg = 0; // 1 = show the "dropped with X attributes" chat messages
	.debug   = 0; // 1 = write the ground drop debugmes line to the server log
	.vm      = callfunc("F_ModSetting", "affix-forge", "value_mult", 100);
	.uniq    = callfunc("F_ModSetting", "affix-forge", "unique_stats", 1);
	.w1      = callfunc("F_ModSetting", "affix-forge", "weight_common", 55);
	.w2      = callfunc("F_ModSetting", "affix-forge", "weight_uncommon", 28);
	.w3      = callfunc("F_ModSetting", "affix-forge", "weight_rare", 14);
	.w4      = callfunc("F_ModSetting", "affix-forge", "weight_epic", 3);
	// "count_weights" is one text setting, "w1,w2,w3,w4": how often an item gets
	// 1, 2, 3 or 4 attributes. A mod may declare only 20 settings, hence one string.
	explode(.@cw$, callfunc("F_ModSetting", "affix-forge", "count_weights", "50,30,15,5"), ",");
	.cwsum = 0;
	for (.@i = 1; .@i <= 4; .@i++) {
		.cw[.@i] = atoi(.@cw$[.@i - 1]);
		if (.cw[.@i] < 0) .cw[.@i] = 0;
		.cwsum += .cw[.@i];
	}
	if (.cwsum < 1) { .cw[1] = 1; .cwsum = 1; }
	.wsum = .w1 + .w2 + .w3 + .w4;
	if (.wsum < 1) { .w1 = 1; .wsum = 1; }

	// The option pool, one row per option:
	//   callsub L_Add, <option id>, <tier 1-4>, <min>, <max>, <kind>, <item type>, <cap>, <group>;
	//   id         rAthena's item_randomopt_db id
	//   tier       rarity: the tier weights in the mod settings decide how often a tier is rolled
	//   min, max   base value range (0, 0 = the option has no value)
	//   kind       0 = flat value, scales with monster level (x(1 + 2/100 per level))
	//              1 = percent value, scales (x(1 + 1/100 per level))
	//              2 = never scales
	//   item type  0 = any item, 5 = weapons only, 4 = armor only
	//   cap        hard ceiling on the rolled value, 0 = none
	//   group      options with the same non-zero group never roll on one item. Groups 1-9 are
	//              always enforced (weapon element, armor element). Groups 10 and up are stat
	//              families (MaxHP and MaxHP %, ATK and ATK %, ...), enforced only while the
	//              "unique_stats" setting is on.

	callsub L_Add, 182, 4, 0, 0, 2, 5, 0, 1;	// Weapon Element: Dark
	callsub L_Add, 183, 4, 0, 0, 2, 5, 0, 1;	// Weapon Element: Ghost
	callsub L_Add, 184, 4, 0, 0, 2, 5, 0, 1;	// Weapon Element: Undead
	callsub L_Add, 76, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Neutral
	callsub L_Add, 77, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Water
	callsub L_Add, 78, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Earth
	callsub L_Add, 79, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Fire
	callsub L_Add, 80, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Wind
	callsub L_Add, 81, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Poison
	callsub L_Add, 82, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Holy
	callsub L_Add, 83, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Dark
	callsub L_Add, 84, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Ghost
	callsub L_Add, 85, 4, 0, 0, 2, 4, 0, 2;	// Armor Element: Undead

	// Index the pool by tier, for the rarity roll.
	for (.@i = 0; .@i < .n; .@i++) {
		.@t = .p_tier[.@i];
		.ti[(.@t - 1) * 400 + .tc[.@t]] = .@i;
		.tc[.@t]++;
	}
	end;

L_Add:
	.p_id[.n]   = getarg(0);
	.p_tier[.n] = getarg(1);
	.p_min[.n]  = getarg(2);
	.p_max[.n]  = getarg(3);
	.p_kind[.n] = getarg(4);
	.p_type[.n] = getarg(5);
	.p_cap[.n]  = getarg(6);
	.p_grp[.n]  = getarg(7);
	.n++;
	return;

OnNPCKillEvent:
	if (.debug) debugmes "affix-forge: kill event mob " + killedrid + " gid " + killedgid + " mult " + .mult + " ground " + .ground;
	if (.mult < 1) end;
	// getmobdrops fills $@MobDrop_item[] / $@MobDrop_rate[] and returns the
	// drop count, or 0 for an unknown monster.
	if (getmobdrops(killedrid) == 0) end;
	.@count = $@MobDrop_count;
	for (.@d = 0; .@d < .@count; .@d++) {
		.@item[.@d] = $@MobDrop_item[.@d];
		.@rate[.@d] = $@MobDrop_rate[.@d];
	}

	.@lv = getmonsterinfo(killedrid, MOB_LV);

	// Where the monster died. getmapxy cannot look up monsters, so read them with
	// getunitdata (killedgid = the dead monster's unit id; UMOB_X / UMOB_Y are the
	// monster fields this server uses). Start from the player's position and only
	// take the monster's if it is a real spot near the player, so a failed lookup
	// falls back to the player.
	getmapxy(.@map$, .@x, .@y, BL_PC);
	if (.debug) debugmes "affix-forge: player at " + .@map$ + " " + .@x + "," + .@y + ", reading monster position";
	getunitdata(killedgid, .@mob);
	if (.debug) debugmes "affix-forge: monster at " + .@mob[UMOB_X] + "," + .@mob[UMOB_Y];
	.@mx = .@mob[UMOB_X];
	.@my = .@mob[UMOB_Y];
	.@dx = .@mx - .@x;
	.@dy = .@my - .@y;
	if (.@dx < 0) .@dx = -.@dx;
	if (.@dy < 0) .@dy = -.@dy;
	if (.@mx > 0 && .@my > 0 && .@dx <= 30 && .@dy <= 30) {
		.@x = .@mx;
		.@y = .@my;
	} else if (.debug) {
		debugmes "affix-forge: could not read the monster position, using the player position";
	}

	.@flat = 100;
	.@pct  = 100;
	if (.scaling) {
		.@flat = 100 + .@lv * 2;
		.@pct  = 100 + .@lv;
	}

	for (.@d = 0; .@d < .@count; .@d++) {
		.@id = .@item[.@d];
		.@type = getiteminfo(.@id, ITEMINFO_TYPE);
		// 4 armor, 5 weapon, 12 shadow gear
		if (.@type != 4 && .@type != 5 && .@type != 12) continue;
		if (rand(10000) >= .@rate[.@d] * .mult / 100) continue;

		// How many attributes, then that many distinct options: each one rolls
		// a tier by weight, then a random option of that tier that fits the item.
		// The count is weighted (setting count_weights).
		.@r = rand(1, .cwsum);
		.@k = 1;
		for (.@j = 1; .@j <= 4; .@j++) {
			.@r -= .cw[.@j];
			if (.@r <= 0) { .@k = .@j; break; }
		}
		.@got = 0;
		cleararray .@oid[0], 0, 4;
		cleararray .@oval[0], 0, 4;
		cleararray .@opar[0], 0, 4;
		cleararray .@ogrp[0], 0, 4;
		for (.@j = 0; .@j < .@k; .@j++) {
			.@s = -1;
			.@tries = 0;
			while (.@s < 0 && .@tries < 80) {
				.@tries++;
				.@r = rand(1, .wsum);
				if (.@r <= .w1) .@t = 1;
				else if (.@r <= .w1 + .w2) .@t = 2;
				else if (.@r <= .w1 + .w2 + .w3) .@t = 3;
				else .@t = 4;
				if (.tc[.@t] < 1) continue;
				.@c = .ti[(.@t - 1) * 400 + rand(.tc[.@t])];
				if (.p_type[.@c] && .p_type[.@c] != .@type) continue;
				.@dup = 0;
				for (.@u = 0; .@u < .@got; .@u++) {
					if (.@oid[.@u] == .p_id[.@c]) .@dup = 1;
					if (.p_grp[.@c] && .@ogrp[.@u] == .p_grp[.@c] && (.p_grp[.@c] < 10 || .uniq)) .@dup = 1;
				}
				if (.@dup) continue;
				.@s = .@c;
			}
			if (.@s < 0) continue;

			.@v = 1;
			if (.p_max[.@s] > 0) {
				.@v = rand(.p_min[.@s], .p_max[.@s]);
				if (.p_kind[.@s] == 0) .@v = .@v * .@flat / 100;
				else if (.p_kind[.@s] == 1) .@v = .@v * .@pct / 100;
				.@v = .@v * .vm / 100;
				if (.p_cap[.@s] > 0 && .@v > .p_cap[.@s]) .@v = .p_cap[.@s];
				if (.@v < 1) .@v = 1;
			}
			.@oid[.@got]  = .p_id[.@s];
			.@oval[.@got] = .@v;
			.@ogrp[.@got] = .p_grp[.@s];
			.@got++;
		}
		if (.@got < 1) continue;
		.@k = .@got;

		// Would the player's @autoloot have picked up the stock drop? Same test
		// as the server: the item is on the @autolootitem list, or its rate (as
		// the player sees it) is within AUTOLOOT_RATE and its type is allowed.
		.@auto = 0;
		if (.ground) {
			if (AUTOLOOT_RATE > 0) {
				.@mine = getmobdroprate(killedrid, .@id);
				if (.@mine > 0 && .@mine <= AUTOLOOT_RATE && (AUTOLOOT_TYPE == 0 || (AUTOLOOT_TYPE & (1 << .@type)))) .@auto = 1;
			}
			if (!.@auto) {
				.@n = getautolootitems(.@list);
				for (.@u = 0; .@u < .@n; .@u++)
					if (.@list[.@u] == .@id) .@auto = 1;
			}
		}

		if (.ground && !.@auto) {
			// Drop it on the ground, unidentified. The last argument asks for
			// the pillar of light (setting ground_pillar); the server picks its color from the option count
			// (battle conf rndopt_drop_pillar, on by default): blue 1, yellow 2,
			// purple 3 or more.
			makeitem3 .@id, 1, .@map$, .@x, .@y, 0, 0, 0, 0, 0, 0, 0, .@oid, .@oval, .@opar, .pillar;
			if (.debug) debugmes "affix-forge: ground drop item " + .@id + " on " + .@map$ + " " + .@x + "," + .@y + " options " + .@oid[0] + "/" + .@oid[1] + "/" + .@oid[2] + "/" + .@oid[3];
			if (.showmsg) dispbottom "A " + getitemname(.@id) + " dropped with " + .@k + " hidden attribute" + (.@k > 1 ? "s" : "") + "!", 0x00FF88;
			continue;
		}

		if (!checkweight(.@id, 1)) {
			dispbottom "You are too heavy to pick up the " + getitemname(.@id) + " the monster dropped!", 0xFF8800;
			continue;
		}
		// Ground drop mode (autoloot): stays unidentified like a ground drop.
		getitem3 .@id, 1, (.ground ? 0 : 1), 0, 0, 0, 0, 0, 0, .@oid, .@oval, .@opar;
		if (.showmsg) dispbottom "A " + getitemname(.@id) + " dropped with " + .@k + (.ground ? " hidden" : " bonus") + " attribute" + (.@k > 1 ? "s" : "") + "!", 0x00FF88;
	}
	end;
}
