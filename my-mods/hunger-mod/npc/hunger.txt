prontera,155,185,4	script	Hunger Guide#hungermod	4_F_KAFRA1,{
    if (hm_initialized == 0) {
        set hm_hunger, 100;
        set hm_initialized, 1;
    }

    mes "[Hunger Guide]";
    mes "Tu hambre: " + hm_hunger + " / 100";
    if (hm_hunger == 0) {
        mes "Estas agotado: tu velocidad y tus seis estadisticas";
        mes "estan reducidas un 20%. Usa alimentos para recuperarte.";
    }
    mes "Al usar alimentos recuperaras hambre.";
    close;
}

-	script	hunger_mgr	-1,{
OnPCLoginEvent:
    if (hm_initialized == 0) {
    	set hm_hunger, 100;
    	set hm_initialized, 1;
    }
    dispbottom "@@event hunger update " + hm_hunger;
    deltimer "hunger_mgr::OnHungerTick";
    deltimer "hunger_mgr::OnPenaltyTick";
    // Testing rate: 1 hunger point every 15 seconds.
    addtimer 15000, "hunger_mgr::OnHungerTick";
    addtimer 1000, "hunger_mgr::OnPenaltyTick";
    end;

OnPCLogoutEvent:
    deltimer "hunger_mgr::OnHungerTick";
    deltimer "hunger_mgr::OnPenaltyTick";
    set @hm_penalty_str, 0;
    set @hm_penalty_agi, 0;
    set @hm_penalty_vit, 0;
    set @hm_penalty_int, 0;
    set @hm_penalty_dex, 0;
    set @hm_penalty_luk, 0;
    end;

OnHungerTick:
    if (hm_hunger > 0)
    	set hm_hunger, hm_hunger - 1;
    dispbottom "@@event hunger update " + hm_hunger;
    addtimer 15000, "hunger_mgr::OnHungerTick";
    end;

OnPenaltyTick:
    if (hm_hunger == 0) {
    	// Add back this mod's last penalty before calculating 20% of each stat.
    	.@str = readparam(bStr) + @hm_penalty_str;
    	.@agi = readparam(bAgi) + @hm_penalty_agi;
    	.@vit = readparam(bVit) + @hm_penalty_vit;
    	.@int = readparam(bInt) + @hm_penalty_int;
    	.@dex = readparam(bDex) + @hm_penalty_dex;
    	.@luk = readparam(bLuk) + @hm_penalty_luk;

    	set @hm_penalty_str, .@str * 20 / 100;
    	set @hm_penalty_agi, .@agi * 20 / 100;
    	set @hm_penalty_vit, .@vit * 20 / 100;
    	set @hm_penalty_int, .@int * 20 / 100;
    	set @hm_penalty_dex, .@dex * 20 / 100;
    	set @hm_penalty_luk, .@luk * 20 / 100;

    	.@penalty$ = "bonus bStr,-" + @hm_penalty_str + ";";
    	.@penalty$ += "bonus bAgi,-" + @hm_penalty_agi + ";";
    	.@penalty$ += "bonus bVit,-" + @hm_penalty_vit + ";";
    	.@penalty$ += "bonus bInt,-" + @hm_penalty_int + ";";
    	.@penalty$ += "bonus bDex,-" + @hm_penalty_dex + ";";
    	.@penalty$ += "bonus bLuk,-" + @hm_penalty_luk + ";";
    	.@penalty$ += "bonus bSpeedRate,-20;";
    	bonus_script .@penalty$, 5, 1032;
    } else {
    	set @hm_penalty_str, 0;
    	set @hm_penalty_agi, 0;
    	set @hm_penalty_vit, 0;
    	set @hm_penalty_int, 0;
    	set @hm_penalty_dex, 0;
    	set @hm_penalty_luk, 0;
    }
    addtimer 4000, "hunger_mgr::OnPenaltyTick";
    end;
}
