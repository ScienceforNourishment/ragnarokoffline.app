//===== Ragnarok Offline: tavern-rumors, alijos =============================
//= El rumor de un alijo (0.2.0): la cadena «Rumor → Pista → Hallazgo» de
//= PLAN.md. El Tabernero vende el rumor; dice el campo y un punto, y el alijo
//= está a 8 celdas o menos de ese punto, a la vista. Quien compró el rumor lo
//= abre: una pieza de equipo sin identificar con afijos del Tier del mapa
//= (las funciones de dungeon-chest) y un relato de adventure.
//=
//= Desde la 0.4.0 (D45) el alijo es de lugares lejanos: un campo del atlas de
//= otra región ($@wa_field$, F_Tavern_Region), del Tier de la banda de nivel
//= + 1 o el más cercano a ese. Con el ajuste find_nearby (las pruebas, D23), o
//= sin atlas, sigue en los campos de la ciudad (F_Tavern_Fields).
//=
//= Un rumor por personaje, sin caducidad: comprar otro descarta el anterior.
//= Es independiente de las cadenas (tavern-chains.txt): variables distintas,
//= así que se puede tener un alijo y una cadena a la vez.
//= Sin temporizadores: el alijo se coloca al entrar en su mapa
//= (OnPCLoadMapEvent), así que vuelve tras reiniciar el servidor.
//=
//= Variables de personaje (permanentes):
//=   tr_hm$          mapa del alijo; "" sin rumor
//=   tr_hx, tr_hy    su celda
//=   tr_px, tr_py    el punto que dijo el Tabernero
//= Servidor:
//=   $@tr_hn$[char_id]  nombre único de la copia de Alijo#tr de ese personaje.
//=                      Tras abrirlo queda oculta (un NPC no puede descargarse
//=                      a sí mismo) y se descarga al comprar el siguiente rumor.
//=
//= Dependencias suaves: dungeon-chest (el botín; sin él, zeny), world-atlas
//= (Tier y nombre del mapa), adventure (la nota del diario y el relato).
//===========================================================================

// F_Tavern_Rand(min, max) -> un entero de min a max, ambos incluidos. rand
// falla con un solo valor y corta el script (CONVENTIONS.md).
function	script	F_Tavern_Rand	{
	if (getarg(1) <= getarg(0))
		return getarg(0);
	return rand(getarg(0), getarg(1));
}

// F_Tavern_FindPrice() -> zeny del rumor de un alijo: find_rounds rondas.
function	script	F_Tavern_FindPrice	{
	return callfunc("F_Tavern_Price") * callfunc("F_ModSetting", "tavern-rumors", "find_rounds", 3);
}

// F_Tavern_FindFar() -> 1 si el alijo va lejos: find_nearby apagado y el atlas
// cargado (sin él no se sabe qué es lejos).
function	script	F_Tavern_FindFar	{
	return $@wa_ready && !callfunc("F_ModSetting", "tavern-rumors", "find_nearby", 1);
}

// F_Tavern_FindMap(ciudad$) -> el campo del próximo alijo del personaje
// adjunto, o "" si no hay ninguno. Lejos: un campo de otra región con el Tier
// de su banda + 1; si no hay, el más cercano a ese Tier, y ante un empate el
// más bajo. Cerca: uno de los campos de la ciudad.
function	script	F_Tavern_FindMap	{
	if (!callfunc("F_Tavern_FindFar")) {
		.@list$ = callfunc("F_Tavern_Fields", getarg(0));
		if (.@list$ == "")
			return "";
		explode(.@f$, .@list$, "|");
		return .@f$[callfunc("F_Tavern_Rand", 0, getarraysize(.@f$) - 1)];
	}
	.@home$ = callfunc("F_Tavern_Region", getarg(0));
	.@want = min(6, callfunc("F_Tavern_Band") + 1);
	.@best = 999;
	.@n = 0;
	for (.@k = 0; .@k < getarraysize($@wa_field$); .@k++) {
		.@m$ = $@wa_field$[.@k];
		if (compare(.@home$, "|" + callfunc("F_Atlas_Anchor", .@m$) + "|"))
			continue;
		.@t = callfunc("F_Atlas_Tier", .@m$);
		// Lo que se pasa del Tier buscado pesa un poco más que lo que se queda corto.
		.@d = (.@t > .@want ? (.@t - .@want) * 2 + 1 : (.@want - .@t) * 2);
		if (.@d > .@best)
			continue;
		if (.@d < .@best) {
			.@best = .@d;
			.@n = 0;
		}
		.@c$[.@n] = .@m$;
		.@n++;
	}
	if (!.@n)
		return "";
	return .@c$[callfunc("F_Tavern_Rand", 0, .@n - 1)];
}

// F_Tavern_NewFind(ciudad$) -> 1 si dejó un rumor nuevo al personaje adjunto
// (en el campo de F_Tavern_FindMap), 0 si no encontró sitio. Descarga el
// alijo anterior, abierto o no.
function	script	F_Tavern_NewFind	{
	.@m$ = callfunc("F_Tavern_FindMap", getarg(0));
	if (.@m$ == "")
		return 0;
	// Sin esto, el alijo de un campo lejano no aparece (OnPCLoadMapEvent).
	setmapflag .@m$, mf_loadevent;
	// Los campos miden 400 o menos; una celda fuera del mapa no es transitable.
	.@ok = 0;
	for (.@try = 0; .@try < 500; .@try++) {
		.@x = rand(20, 380);
		.@y = rand(20, 380);
		if (checkcell(.@m$, .@x, .@y, cell_chkpass)) {
			.@ok = 1;
			break;
		}
	}
	if (!.@ok) {
		debugmes "tavern-rumors: no passable cell on " + .@m$ + " after 500 attempts";
		return 0;
	}
	.@cid = getcharid(0);
	if ($@tr_hn$[.@cid] != "") {
		if (getnpcid(0, $@tr_hn$[.@cid]))
			unloadnpc $@tr_hn$[.@cid];
		$@tr_hn$[.@cid] = "";
	}
	tr_hm$ = .@m$;
	tr_hx = .@x;
	tr_hy = .@y;
	tr_px = .@x + rand(-8, 8);
	tr_py = .@y + rand(-8, 8);
	if (tr_px < 1)
		tr_px = 1;
	if (tr_py < 1)
		tr_py = 1;
	return 1;
}

// F_Tavern_FindText() -> la pista del rumor del personaje adjunto, en
// minúscula: "un alijo en <lugar> (por <ciudad>), a pocos pasos de (x, y)."
function	script	F_Tavern_FindText	{
	.@by$ = "";
	if ($@wa_ready) {
		.@a$ = callfunc("F_Atlas_Anchor", tr_hm$);
		if (.@a$ != "")
			.@by$ = " (por " + callfunc("F_Tavern_Place", .@a$) + ")";
	}
	return "un alijo en " + callfunc("F_Tavern_Place", tr_hm$) + .@by$ + ", a pocos pasos de (" + tr_px + ", " + tr_py + ").";
}

// La mapflag loadevent de los campos lejanos se pierde al reiniciar: se vuelve
// a poner en todos los del atlas. El atlas se llena en su propio OnInit, sin
// orden garantizado, así que se le espera unos segundos.
OnInit:
	for (.@t = 0; .@t < 30 && !$@wa_ready; .@t++)
		sleep 1000;
	for (.@k = 0; .@k < getarraysize($@wa_field$); .@k++)
		setmapflag $@wa_field$[.@k], mf_loadevent;
	end;

// Coloca el alijo del personaje al entrar en su mapa, si no está ya.
OnPCLoadMapEvent:
	if (tr_hm$ == "" || tr_hm$ != strcharinfo(3))
		end;
	.@cid = getcharid(0);
	if ($@tr_hn$[.@cid] != "" && getnpcid(0, $@tr_hn$[.@cid]))
		end;
	// 3075 = WA_TREASURE, el sprite de los cofres de dungeon-chest. Un NPC
	// flotante no tiene sprite propio: se pasa a duplicate (campfire).
	.@npc$ = duplicate("Alijo#tr", tr_hm$, tr_hx, tr_hy, "Alijo", 3075);
	if (.@npc$ == "") {
		debugmes "tavern-rumors: could not place the find on " + tr_hm$ + " " + tr_hx + "," + tr_hy;
		end;
	}
	$@tr_hn$[.@cid] = .@npc$;
	end;
}

		else

	tr_hy = 0;

	tr_py = 0;

	close2;

		callfunc "F_Adv_Note", "note", "Encontré el alijo del rumor en " + .@name$ + ".";
		callfunc "F_Adv_Experience", "rumor:" + .@m$, 1, .@tier, "Encontré el alijo del rumor en " + .@name$, 1;
	}
	end;
}
