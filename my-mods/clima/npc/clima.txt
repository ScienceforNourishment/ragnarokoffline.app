OnTimer1000:
	.@now = gettimetick(2);

	for (.@i = 0; .@i < .map_count; .@i++) {
		if (.event_active[.@i] && .@now >= .event_end[.@i]) {
			.event_active[.@i] = 0;
			.event_end[.@i] = 0;
			if (.next_check[.@i] <= .@now)
				.next_check[.@i] = .@now + .check_interval;
			debugmes "[clima] Evento " + .combo_name$[.combination[.@i]] + " terminado en " + .map$[.@i] + ".";
			mapannounce .map$[.@i], .combo_end_message$[.combination[.@i]], bc_map;
			.@player_count = getmapunits(BL_PC, .map$[.@i], .@player_ids[0]);
			freeloop(1);
			for (.@j = 0; .@j < .@player_count; .@j++) {
				callsub S_ApplyWeatherStatus, .@i, .@player_ids[.@j];
			}
			freeloop(0);
		}

		if (.event_active[.@i] && .@now >= .next_effect_refresh[.@i]) {
			.next_effect_refresh[.@i] = .@now + .effect_refresh_interval;
			.@player_count = getmapunits(BL_PC, .map$[.@i], .@player_ids[0]);
			freeloop(1);
			for (.@j = 0; .@j < .@player_count; .@j++) {
				callsub S_ApplyWeatherStatus, .@i, .@player_ids[.@j];
			}
			freeloop(0);
		}

		if (!.event_active[.@i] && .@now >= .next_check[.@i]) {
			.next_check[.@i] = .@now + .check_interval;
			if (.chance[.@i] >= 100 || rand(100) < .chance[.@i]) {
				.event_active[.@i] = 1;
				.event_end[.@i] = .@now + .duration[.@i];
				.next_effect_refresh[.@i] = .@now + .effect_refresh_interval;
				.@player_count = getmapunits(BL_PC, .map$[.@i], .@player_ids[0]);
				freeloop(1);
				for (.@j = 0; .@j < .@player_count; .@j++) {
					callsub S_ApplyWeatherStatus, .@i, .@player_ids[.@j];
				}
				freeloop(0);
				debugmes "[clima] Evento " + .combo_name$[.combination[.@i]] + " iniciado en " + .map$[.@i] + "; visual " + .combo_weather$[.combination[.@i]] + "; estado " + .combo_status_name$[.combination[.@i]] + "; personajes afectados: " + .@player_count + ".";
				mapannounce .map$[.@i], .combo_start_message$[.combination[.@i]], bc_map;
			} else {
				debugmes "[clima] Revisión de " + .map$[.@i] + ": no comenzó " + .combo_name$[.combination[.@i]] + " (probabilidad " + .chance[.@i] + "%).";
			}
		}
	}

	initnpctimer;
	end;

S_IsWeatherProtected:
	.@map_index = getarg(0);
	.@combo_id = .combination[.@map_index];
	.@item_id = .combo_protection_item[.@combo_id];
	if (!.@item_id)
		return 0;
	if (getequipid(.combo_protection_slot[.@combo_id]) == .@item_id)
		return 1;
	if (.combo_protection_slot_alt[.@combo_id] >= 0) {
		if (getequipid(.combo_protection_slot_alt[.@combo_id]) == .@item_id)
			return 1;
	}
	return 0;

S_ApplyWeatherStatus:
	.@map_index = getarg(0);
	.@player_id = getarg(1);
	if (!attachrid(.@player_id)) {
		debugmes "[clima] No se pudo adjuntar al personaje " + .@player_id + " para revisar su estado.";
		return;
	}
	.@combo_id = .combination[.@map_index];
	if (!.event_active[.@map_index]) {
		set @clima_hunger_extra_loss, 0;
		set @clima_thirst_extra_loss, 0;
		set @clima_exposure_map$, "";
		detachrid;
		return;
	}

	.@protected = callsub(S_IsWeatherProtected, .@map_index);
	set @clima_hunger_extra_loss, 0;
	set @clima_thirst_extra_loss, 0;
	set @clima_exposure_map$, "";
	if (!.@protected && .@combo_id == 9) {
		set @clima_thirst_extra_loss, 1;
		set @clima_exposure_map$, .map$[.@map_index];
	} else if (!.@protected && .@combo_id == 10) {
		set @clima_hunger_extra_loss, 1;
		set @clima_exposure_map$, .map$[.@map_index];
	}
	if (.combo_status[.@combo_id] && !.@protected && !getstatus(.combo_status[.@combo_id]))
		sc_start .combo_status[.@combo_id], .combo_status_duration[.@combo_id] * 1000, 0, 10000, SCSTART_NOAVOID;
	detachrid;
	return;

OnPCLoadMapEvent:
	set @clima_hunger_extra_loss, 0;
	set @clima_thirst_extra_loss, 0;
	set @clima_exposure_map$, "";
	.@map$ = strcharinfo(3);
	for (.@i = 0; .@i < .map_count; .@i++) {
		if (.@map$ == .map$[.@i] && .event_active[.@i]) {
			.@remaining = (.event_end[.@i] - gettimetick(2)) * 1000;
			if (.@remaining > 0) {
				.@combo_id = .combination[.@i];
				.@lease = min(.combo_status_duration[.@combo_id] * 1000, .@remaining);
				.@protected = callsub(S_IsWeatherProtected, .@i);
				if (.combo_status[.@combo_id] && !.@protected && !getstatus(.combo_status[.@combo_id]))
					sc_start .combo_status[.@combo_id], .@lease, 0, 10000, SCSTART_NOAVOID;
				if (.combo_status[.@combo_id])
					.@status_active = getstatus(.combo_status[.@combo_id]);
				else
					.@status_active = 0;
				debugmes "[clima] " + .combo_status_name$[.@combo_id] + " al entrar a " + .@map$ + "; activo: " + .@status_active + "; protegido: " + .@protected + "; lease: " + .@lease + " ms.";
				if (!.@protected && .@combo_id == 9) {
					set @clima_thirst_extra_loss, 1;
					set @clima_exposure_map$, .@map$;
				} else if (!.@protected && .@combo_id == 10) {
					set @clima_hunger_extra_loss, 1;
					set @clima_exposure_map$, .@map$;
				}
			}
			break;
		}
	}
	end;
