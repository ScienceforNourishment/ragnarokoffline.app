//===== Ragnarok Offline: nemesis ==========================================
//= El monstruo que te mata fuera de una ciudad se convierte en tu nemesis: la
//= proxima vez que entres en ese mapa te espera, con nombre y mas fuerte, cerca
//= de donde caiste. Si vuelve a matarte sube de rango (hasta 3). Vencerlo da
//= un talisman firmado con tu nombre, un relato de adventure y cura las heridas
//= que te dejo (wounds).
//=
//= El talisman (D40) es un accesorio de su raza (51200 + raza) cuyos afijos
//= cuentan su historia: ATK y MATK +(talisman_base + talisman_rank x rango) %
//= contra su raza (12/14/16 por defecto) y, si te dejo una herida, el
//= atributo que esa herida te quitaba (S_WoundAffix). Tambien se cuelga con el
//= Coleccionista de expedition-board o se engarza con el Herrero de
//= dungeon-chest; no sirve de prueba para las marcas de elite.
//=
//= Un solo nemesis a la vez: mientras tengas uno, morir ante otro monstruo no
//= crea otro. No cuentan los MVP, las marcas de expedition-board ni el nemesis
//= de otro jugador.
//=
//= Estado por personaje (variables permanentes):
//=   nm_mob        clase del monstruo; 0 = sin nemesis
//=   nm_map$       mapa donde te espera
//=   nm_x, nm_y    celda donde caiste; aparece cerca
//=   nm_rank       1 a 3; sube cada vez que vuelve a matarte
//=   nm_owed       talisman que no cupo (id del objeto); nm_owed_n el rango del
//=                 nemesis y nm_owed_w la herida que dejo (0 = ninguna). Antes
//=                 de la 0.3, nm_owed_n era cuantos trofeos: era el mismo numero
//= Estado del servidor (se pierde al reiniciar, a proposito):
//=   $@nm_gid[char_id]  GID del nemesis vivo de ese personaje
//=   $@nm_owner[gid]    char_id del dueno de ese nemesis
//=
//= Sin temporizadores: aparece con OnPCLoadMapEvent, que solo corre en mapas
//= con la mapflag loadevent. Se la pone el propio mod a su mapa (setmapflag)
//= al nacer el nemesis y al entrar el personaje, porque tras un reinicio se
//= pierde. Ver CONVENTIONS.md, Temporizadores.
//=
//= Dependencias suaves (el mod funciona sin ellas):
//=   world-atlas  ciudades, nombre del lugar y Tier (refuerzo y relato)
//=   wounds       F_Wounds_CureFrom: cura las heridas que dejo esa clase
//=   adventure    relato "nemesis:<clase>" al vencerlo
//=   adventure    notas en el diario (F_Adv_Note, adventure 0.3)
//===========================================================================

// F_Nemesis_Name(clase, rango) -> nombre del nemesis, 23 caracteres como
// mucho (NAME_LENGTH de rAthena); si no cabe, se acorta el del monstruo.
function	script	F_Nemesis_Name	{
	.@rank = getarg(1);
	.@t$ = "el Verdugo";
	if (.@rank == 2)
		.@t$ = "el Implacable";
	if (.@rank >= 3)
		.@t$ = "el Eterno";
	.@m$ = getmonsterinfo(getarg(0), MOB_NAME);
	.@room = 23 - getstrlen(.@t$) - 1;
	if (getstrlen(.@m$) > .@room)
		.@m$ = substr(.@m$, 0, .@room - 1);
	return .@m$ + " " + .@t$;
}

// F_Nemesis_Alive() -> 1 si el nemesis del personaje esta en pie.
function	script	F_Nemesis_Alive	{
	.@gid = $@nm_gid[getcharid(0)];
	return (.@gid > 0 && unitexists(.@gid));
}

// F_Nemesis_Place(mapa$) -> nombre del lugar segun el atlas, o el codigo.
function	script	F_Nemesis_Place	{
	if (!$@wa_ready)
		return getarg(0);
	.@n$ = callfunc("F_Atlas_Name", getarg(0));
	if (.@n$ == "")
		return getarg(0);
	return .@n$;
}

// F_Nemesis_Log(texto$) -> anota una linea en el diario del viajero, si esta.
// F_Adv_Note (adventure 0.3) no la pierde aunque el personaje este en un dialogo.
function	script	F_Nemesis_Log	{
	if (getnpcid(0, "Adventure"))
		callfunc "F_Adv_Note", "note", getarg(0);
	return;
}

// F_Nemesis_Retire() -> retira del mapa el nemesis vivo del personaje, sin
// premio: se olvida su GID antes de matarlo, asi OnNemesisDead lo ignora.
function	script	F_Nemesis_Retire	{
	.@cid = getcharid(0);
	.@gid = $@nm_gid[.@cid];
	$@nm_gid[.@cid] = 0;
	if (.@gid < 1)
		return;
	$@nm_owner[.@gid] = 0;
	if (unitexists(.@gid))
		unitkill .@gid;
	return;
}

-	script	nemesis_mgr	-1,{
	end;

OnInit:
	// Ajustes (mod.json): el rango maximo y cuanto refuerza cada rango.
	.max_rank = callfunc("F_ModSetting", "nemesis", "max_rank", 3);
	.rank_hp = callfunc("F_ModSetting", "nemesis", "rank_hp", 50);
	.rank_atk = callfunc("F_ModSetting", "nemesis", "rank_atk", 20);
	.tal_base = callfunc("F_ModSetting", "nemesis", "talisman_base", 10);
	.tal_rank = callfunc("F_ModSetting", "nemesis", "talisman_rank", 2);
	// Sin atlas, estas son las ciudades (la misma lista que wounds).
	.safe_maps$ = "|prontera|izlude|geffen|payon|morocc|alberta|aldebaran|";
	bindatcmd "nemesis", "nemesis_mgr::OnShow", 0, 99;
	bindatcmd "nemesisgm", "nemesis_mgr::OnGM", 1, 99;
	end;

OnPCLoginEvent:
	if (nm_owed)
		callsub S_GiveOwed;
	if (!nm_mob)
		end;
	setmapflag nm_map$, mf_loadevent;
	if (strcharinfo(3) == nm_map$)
		callsub S_Spawn;
	end;

// Corre en cada cambio de mapa con loadevent, de todos los personajes: la
// primera linea descarta a quien no tiene nemesis.
OnPCLoadMapEvent:
	if (!nm_mob)
		end;
	if (strcharinfo(3) != nm_map$)
		end;
	callsub S_Spawn;
	end;

OnPCDieEvent:
	.@map$ = strcharinfo(3);
	// Ni en ciudad ni en instancias (1@xxx).
	if ($@wa_ready) {
		if (callfunc("F_Atlas_IsSafe", .@map$))
			end;
	}
	if (compare(.safe_maps$, "|" + .@map$ + "|"))
		end;
	if (charat(.@map$, 1) == "@")
		end;
	.@killer = killerrid;
	if (!.@killer)
		end;
	if (getunittype(.@killer) != BL_MOB)
		end;
	.@cid = getcharid(0);
	if (.@killer == $@nm_gid[.@cid]) {
		callsub S_RankUp;
		end;
	}
	// Uno a la vez: el primero se queda.
	if (nm_mob)
		end;
	// Ni el nemesis de otro ni la marca de un contrato (expedition-board).
	if ($@nm_owner[.@killer])
		end;
	if ($@eb_markowner[.@killer])
		end;
	getunitdata .@killer, .@ud;
	.@cls = .@ud[UMOB_CLASS];
	if (getmonsterinfo(.@cls, MOB_MODE) & MD_MVP)
		end;
	getmapxy(.@m$, .@x, .@y);
	nm_mob = .@cls;
	nm_map$ = .@map$;
	nm_x = .@x;
	nm_y = .@y;
	nm_rank = 1;
	setmapflag .@map$, mf_loadevent;
	.@mob$ = getmonsterinfo(.@cls, MOB_NAME);
	.@name$ = callfunc("F_Nemesis_Name", .@cls, 1);
	dispbottom "Has caido ante " + .@mob$ + ". No te olvidara: desde ahora es " + .@name$ + ", y te espera en " + callfunc("F_Nemesis_Place", .@map$) + ".";
	callfunc "F_Nemesis_Log", .@mob$ + " me mato en " + callfunc("F_Nemesis_Place", .@map$) + ". Desde hoy es mi nemesis: " + .@name$ + ".";
	end;

// El nemesis vuelve a matarte: sube de rango y se retira; reaparece con su
// nuevo nombre y mas fuerte la proxima vez que entres en el mapa.
S_RankUp:
	if (nm_rank >= .max_rank) {
		dispbottom callfunc("F_Nemesis_Name", nm_mob, nm_rank) + " vuelve a vencerte.";
		return;
	}
	nm_rank++;
	callfunc "F_Nemesis_Retire";
	.@name$ = callfunc("F_Nemesis_Name", nm_mob, nm_rank);
	dispbottom "Tu nemesis se hace mas fuerte con tu caida: ahora es " + .@name$ + ".";
	callfunc "F_Nemesis_Log", "Mi nemesis volvio a vencerme. Ahora lo llaman " + .@name$ + ".";
	return;

// Pone al nemesis cerca de donde caiste, si no esta ya en pie.
S_Spawn:
	if (callfunc("F_Nemesis_Alive"))
		return;
	.@map$ = nm_map$;
	.@sx = 0;
	.@sy = 0;
	for (.@r = 3; .@r <= 15 && .@sx == 0; .@r += 3) {
		for (.@t = 0; .@t < 12; .@t++) {
			.@nx = nm_x - .@r + rand(0, .@r * 2);
			.@ny = nm_y - .@r + rand(0, .@r * 2);
			if (checkcell(.@map$, .@nx, .@ny, cell_chkpass)) {
				.@sx = .@nx;
				.@sy = .@ny;
				break;
			}
		}
	}
	// Sin celda libre cerca: en cualquier punto del mapa (0,0).
	.@name$ = callfunc("F_Nemesis_Name", nm_mob, nm_rank);
	monster .@map$, .@sx, .@sy, .@name$, nm_mob, 1, "nemesis_mgr::OnNemesisDead";
	.@gid = $@mobid[0];
	if (.@gid < 1) {
		debugmes "nemesis: monster() failed for " + nm_mob + " on " + .@map$;
		return;
	}
	$@nm_gid[getcharid(0)] = .@gid;
	$@nm_owner[.@gid] = getcharid(0);
	callsub S_Strengthen, .@gid;
	dispbottom "Algo te esperaba: " + .@name$ + " ronda cerca de donde caiste.";
	return;

// Refuerzo con el Tier del mapa (como las marcas de expedition-board) y el
// rango. Tier T (1 sin atlas), rango R:
//   HP  = (150% + 50% por Tier sobre el 1) x (100% + .rank_hp% por rango sobre el 1)
//   ATK = (100% + 15% por Tier sobre el 1) x (100% + .rank_atk% por rango sobre el 1)
// (.rank_hp 50 y .rank_atk 20 por defecto; ajustes de mod.json.)
// T1 R1: HP x1,5, ATK x1. T1 R3: HP x3, ATK x1,4. T6 R3: HP x8, ATK x2,45.
S_Strengthen:
	.@gid = getarg(0);
	// getunitdata sobre algo que no es un monstruo tumba el map-server.
	if (getunittype(.@gid) != BL_MOB)
		return;
	.@tier = 0;
	if ($@wa_ready)
		.@tier = callfunc("F_Atlas_Tier", nm_map$);
	if (.@tier < 1)
		.@tier = 1;
	.@hp_pct = (150 + 50 * (.@tier - 1)) * (100 + .rank_hp * (nm_rank - 1)) / 100;
	.@atk_pct = (100 + 15 * (.@tier - 1)) * (100 + .rank_atk * (nm_rank - 1)) / 100;
	getunitdata .@gid, .@u;
	.@maxhp = .@u[UMOB_MAXHP] * .@hp_pct / 100;
	setunitdata .@gid, UMOB_MAXHP, .@maxhp;
	setunitdata .@gid, UMOB_HP, .@maxhp;
	setunitdata .@gid, UMOB_ATKMIN, .@u[UMOB_ATKMIN] * .@atk_pct / 100;
	setunitdata .@gid, UMOB_ATKMAX, .@u[UMOB_ATKMAX] * .@atk_pct / 100;
	return;

// Corre con quien mato al nemesis. Sin jugador (se retiro por script) no hay
// nada. Se acredita al dueno: al propio killer o a su companero de grupo si
// esta conectado y en el mismo mapa (CONVENTIONS.md, Regla comun de grupo).
OnNemesisDead:
	if (!playerattached())
		end;
	.@gid = killedgid;
	.@owner = $@nm_owner[.@gid];
	if (.@owner < 1)
		end;
	$@nm_owner[.@gid] = 0;
	if (.@owner != getcharid(0)) {
		.@map$ = strcharinfo(3);
		.@pid = getcharid(1);
		if (!.@pid)
			end;
		.@aid = callsub(S_PartyAid, .@pid, .@owner);
		if (!.@aid)
			end;
		if (!attachrid(.@aid))
			end;
		if (strcharinfo(3) != .@map$)
			end;
	}
	// Desde aqui el personaje adjunto es el dueno.
	if ($@nm_gid[getcharid(0)] != .@gid)
		end;
	$@nm_gid[getcharid(0)] = 0;
	callsub S_Defeat;
	end;

// Premio y cierre: el talisman, cura de sus heridas y relato.
S_Defeat:
	.@cls = nm_mob;
	.@rank = nm_rank;
	.@map$ = nm_map$;
	.@name$ = callfunc("F_Nemesis_Name", .@cls, .@rank);
	nm_mob = 0;
	nm_map$ = "";
	nm_x = 0;
	nm_y = 0;
	nm_rank = 0;
	dispbottom "Has vencido a tu nemesis, " + .@name$ + ".";
	.@race = getmonsterinfo(.@cls, MOB_RACE);
	if (.@race < 0 || .@race > 9)
		.@race = 0;
	nm_owed = 51200 + .@race;
	nm_owed_n = .@rank;
	// La primera herida que te dejo esta clase (wounds la apunta en wd_src$
	// como "herida:clase|..."), antes de que su caida la cure.
	nm_owed_w = 0;
	if (wd_src$ != "") {
		explode(.@s$, wd_src$, "|");
		for (.@i = 0; .@i < getarraysize(.@s$) && !nm_owed_w; .@i++) {
			explode(.@p$, .@s$[.@i], ":");
			if (getarraysize(.@p$) >= 2 && atoi(.@p$[1]) == .@cls)
				nm_owed_w = atoi(.@p$[0]);
		}
	}
	callsub S_GiveOwed;
	if (getnpcid(0, "wounds_mgr")) {
		.@cured = callfunc("F_Wounds_CureFrom", .@cls);
		if (.@cured)
			dispbottom "Con su caida se cierra" + (.@cured > 1 ? "n las heridas" : " la herida") + " que te dejo.";
	}
	if (getnpcid(0, "Adventure")) {
		.@tier = 0;
		if ($@wa_ready)
			.@tier = callfunc("F_Atlas_Tier", .@map$);
		// Mayor; un nemesis de rango 3 es un Hito.
		.@size = 3;
		if (.@rank >= 3)
			.@size = 4;
		callfunc "F_Adv_Experience", "nemesis:" + .@cls, .@size, .@tier, "Venci a mi nemesis, " + .@name$;
	}
	callfunc "F_Nemesis_Log", "Venci a mi nemesis, " + .@name$ + ", en " + callfunc("F_Nemesis_Place", .@map$) + ".";
	return;

// Entrega el talisman pendiente (nm_owed, de rango nm_owed_n y con la herida
// nm_owed_w) si cabe: firmado (card0 254 y el char_id en card2 y card3, como
// getnameditem) y con sus afijos.
S_GiveOwed:
	if (!checkweight(nm_owed, 1)) {
		dispbottom "No puedes cargar el talisman de tu nemesis. Haz sitio y escribe " + ($@nm_sym$ == "" ? "@" : $@nm_sym$) + "nemesis.";
		return;
	}
	.@item = nm_owed;
	.@rank = max(1, nm_owed_n);
	.@race = .@item - 51200;
	.@v = .tal_base + .tal_rank * .@rank;
	// ATK y MATK +% contra la raza: opciones 97 y 107 + raza (affixes.csv).
	setarray .@oid[0], 97 + .@race, 107 + .@race;
	setarray .@oval[0], .@v, .@v;
	setarray .@opar[0], 0, 0;
	callsub S_WoundAffix, nm_owed_w, .@rank;
	if (@nm_wo > 0) {
		.@oid[2] = @nm_wo;
		.@oval[2] = @nm_wv;
		.@opar[2] = 0;
	}
	nm_owed = 0;
	nm_owed_n = 0;
	nm_owed_w = 0;
	.@cid = getcharid(0);
	getitem3 .@item, 1, 1, 0, 0, 254, 0, .@cid & 0xFFFF, (.@cid >> 16) & 0xFFFF, .@oid, .@oval, .@opar;
	dispbottom "Recibes " + getitemname(.@item) + ", un talisman con tu nombre grabado: +" + .@v + "% de dano contra su raza" + (@nm_wo > 0 ? " y algo de la herida que te dejo" : "") + ".";
	return;

// S_WoundAffix(herida, rango): deja en @nm_wo y @nm_wv el afijo que esa herida
// de wounds da al talisman: el atributo que quitaba, +1 + rango (los de % y
// FLEE, mas). 0 si no hay herida o no se conoce.
S_WoundAffix:
	@nm_wo = 0;
	@nm_wv = 0;
	.@w = getarg(0);
	.@r = getarg(1);
	// Por herida (1-10 de wounds): opcion de affixes.csv y valor.
	setarray .@opt[1], 4, 5, 6, 7, 9, 10, 8, 3, 22, 4;
	setarray .@val[1], 1 + .@r, 1 + .@r, 1 + .@r, 1 + .@r, 2 * .@r, 2 * .@r, 1 + .@r, 1 + .@r, 3 * .@r, 1 + .@r;
	if (.@w < 1 || .@w > 10)
		return;
	@nm_wo = .@opt[.@w];
	@nm_wv = .@val[.@w];
	return;

// S_PartyAid(party id, char id) -> account id de ese miembro si esta
// conectado; 0 si no. La misma busqueda que F_Expedition_PartyAid.
S_PartyAid:
	getpartymember getarg(0), 1;
	.@n = $@partymembercount;
	if (.@n < 1)
		return 0;
	copyarray .@cid[0], $@partymembercid[0], .@n;
	getpartymember getarg(0), 2;
	copyarray .@aid[0], $@partymemberaid[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@cid[.@i] == getarg(1))
			return .@aid[.@i];
	}
	return 0;

// @nemesis: quien te espera y donde; entrega un talisman pendiente.
OnShow:
	// El prefijo de comandos (Ajustes -> Rates -> Command prefix puede hacerlo
	// "!" o "."), guardado para el aviso de S_GiveOwed, que tambien sale tras
	// una muerte. Es del servidor, no del jugador; "#" es el de otro personaje.
	if (charat(.@atcmd_command$, 0) != "#")
		$@nm_sym$ = charat(.@atcmd_command$, 0);
	if (nm_owed)
		callsub S_GiveOwed;
	if (!nm_mob) {
		dispbottom "No tienes nemesis. Nadie te espera ahi fuera.";
		end;
	}
	.@name$ = callfunc("F_Nemesis_Name", nm_mob, nm_rank);
	if (callfunc("F_Nemesis_Alive"))
		dispbottom "Tu nemesis: " + .@name$ + " (rango " + nm_rank + "), suelto en " + callfunc("F_Nemesis_Place", nm_map$) + ".";
	else
		dispbottom "Tu nemesis: " + .@name$ + " (rango " + nm_rank + "). Te espera en " + callfunc("F_Nemesis_Place", nm_map$) + ".";
	end;

// @nemesisgm <clase> [rango]: para pruebas (solo GM). Hace nemesis a esa
// clase aqui mismo y lo pone a tu lado. @nemesisgm 0 lo quita.
OnGM:
	.@cls = atoi(.@atcmd_parameters$[0]);
	callfunc "F_Nemesis_Retire";
	if (.@cls == 0) {
		nm_mob = 0;
		nm_map$ = "";
		nm_rank = 0;
		dispbottom "Nemesis quitado.";
		end;
	}
	if (getmonsterinfo(.@cls, MOB_NAME) == "null") {
		dispbottom "No existe el monstruo " + .@cls + ".";
		end;
	}
	.@rank = 1;
	if (.@atcmd_numparameters > 1)
		.@rank = atoi(.@atcmd_parameters$[1]);
	if (.@rank < 1)
		.@rank = 1;
	if (.@rank > .max_rank)
		.@rank = .max_rank;
	getmapxy(.@m$, .@x, .@y);
	nm_mob = .@cls;
	nm_map$ = .@m$;
	nm_x = .@x;
	nm_y = .@y;
	nm_rank = .@rank;
	setmapflag .@m$, mf_loadevent;
	callsub S_Spawn;
	end;
}
