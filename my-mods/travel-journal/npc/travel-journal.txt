//===== Ragnarok Offline: travel-journal ==================================
//= Records what happens to a character in the mod store (scope "char"):
//=   client.n         how many entries were ever written
//=   client.e.<n>     "<unix time>|<type>|<text>", the last 300 are kept
//=   visited.<map>    1 once the character has been on that map
//= Entries under "client" are what the window (client/index.js) reads.
//=
//= Other mods add an entry with the player attached:
//=   @tj_type$ = "note"; @tj_text$ = "Texto sin barras verticales";
//=   doevent "TravelJournal::OnLog";
//= Types the window knows: map, death, note. Any other type is shown as a note.
//= If this mod is off, doevent only logs that the event does not exist.
//=
//= Maps are checked by one clock for every player (initnpctimer + attachrid),
//= not by an addtimer chain per player: a player's own events wait in a 2-slot
//= queue during NPC dialogs and the rest are dropped. This mod's own entries
//= are written with callsub for the same reason. See CONVENTIONS.md,
//= Temporizadores.
//===========================================================================
-	script	TravelJournal	-1,{
	end;

OnInit:
	bindatcmd "diario", strnpcinfo(3) + "::OnCommand", 0, 99;
	initnpctimer;
	end;

OnCommand:
	dispbottom "@@event journal open";
	end;

OnPCLoginEvent:
	@tj_map$ = "";
	callsub S_Join;
	end;

OnPCLogoutEvent:
	callsub S_Leave;
	@tj_map$ = "";
	@tj_type$ = "";
	@tj_text$ = "";
	end;

// Every 5 s, for each player in the list; the clock is restarted first, so a
// script error further down cannot stop it. The first pass after login also
// records the town the character starts in.
OnTimer5000:
	initnpctimer;
	for (.@i = .pl_n - 1; .@i >= 0; .@i--) {
		if (!attachrid(.pl[.@i])) {
			callsub S_Drop, .@i;
			continue;
		}
		callsub S_MapCheck;
		detachrid;
	}
	end;

S_MapCheck:
	.@map$ = strcharinfo(3);
	if (.@map$ == @tj_map$)
		return;
	@tj_map$ = .@map$;
	if (modstore_get("char", "visited." + .@map$, 0) == 0) {
		modstore_set "char", "visited." + .@map$, 1;
		// A map name the store refuses as a key is not written, and so not logged.
		if (modstore_get("char", "visited." + .@map$, 0) == 1)
			callsub S_Write, "map", callsub(S_Place, .@map$);
	}
	return;

OnPCDieEvent:
	callsub S_Write, "death", callsub(S_Place, strcharinfo(3));
	end;

// The readable name of a map from the world-atlas mod; the map code when that
// mod is off or does not know the map. The atlas is a soft dependency: it sets
// $@wa_ready once its table is filled.
S_Place:
	.@code$ = getarg(0);
	if ($@wa_ready) {
		.@name$ = callfunc("F_Atlas_Name", .@code$);
		if (.@name$ != "")
			return .@name$;
	}
	return .@code$;

// Entry point for other mods (see the header).
OnLog:
	if (@tj_type$ == "" || @tj_text$ == "")
		end;
	callsub S_Write, @tj_type$, @tj_text$;
	@tj_type$ = "";
	@tj_text$ = "";
	end;

// S_Write(type$, text$): appends one entry for the attached character.
S_Write:
	.@type$ = replacestr(getarg(0), "|", "/");
	.@text$ = replacestr(getarg(1), "|", "/");
	if (.@type$ == "" || .@text$ == "")
		return;
	.@n = modstore_inc("char", "client.n", 1);
	modstore_set "char", "client.e." + .@n, gettimetick(2) + "|" + .@type$ + "|" + .@text$;
	if (.@n > 300)
		modstore_delete "char", "client.e." + (.@n - 300);
	dispbottom "@@event journal new";
	return;

// The clock's players, by account id. Population Engine bots never fire
// OnPCLoginEvent, so they never join.
S_Join:
	.@aid = getcharid(3);
	for (.@i = 0; .@i < .pl_n; .@i++)
		if (.pl[.@i] == .@aid)
			return;
	.pl[.pl_n] = .@aid;
	.pl_n++;
	return;

S_Leave:
	.@aid = getcharid(3);
	for (.@i = 0; .@i < .pl_n; .@i++) {
		if (.pl[.@i] == .@aid) {
			callsub S_Drop, .@i;
			return;
		}
	}
	return;

// Removes position getarg(0) by moving the last one into it.
S_Drop:
	.@i = getarg(0);
	.pl_n--;
	.pl[.@i] = .pl[.pl_n];
	.pl[.pl_n] = 0;
	return;
}
